// RS VR melee (zscript/rsvr/melee.zs): the axe and the Dragonslayer.

// How hard a swing has to be before the blade cuts. 1.0 is the shipped feel; higher asks for a
// harder swing, lower cuts on less. Multiplies the per-weapon threshold.
user float rsvr_melee_swing = 1.0;
// How much of the blade's motion must point DOWN for a swing to cut: 0.25 is about 15 degrees
// below level (owner, 2026-10-16: only a meaningful downward swing cuts; never level, never up).
user float rsvr_melee_down = 0.25;

// What you HOLD to ready the axe throw, as a usercmd button mask. Default 16 = BT_ALTATTACK,
// which is how it has always worked. QuestZDoom has no usercmd bit for the weapon hand's grip,
// so to throw on the grip either bind the grip to one of the engine's user buttons and set this
// to it (BT_USER1 2097152, BT_USER2 4194304, BT_USER3 8388608, BT_USER4 16777216), or put the
// grip's raw key codes in rsvr_axe_keys below. 0 = never hold, so the throw is off.
user int rsvr_axe_hold = 16;
// Raw key codes that also ready the throw, space separated, read the way rsr_grip_keys is.
// Empty by default. Turn on Print each reload step (rsr_debug) and press your grip to read its
// code, then put it here.
user string rsvr_axe_keys = "";

// Two-handed axe: how squarely the cutting edge must lead the swing before the second
// hand is worth anything. 0.5 is about 60 degrees off the edge; 1.0 demands a perfect
// edge-on blow, 0 pays for any direction. The axe is bladed on one side, so holding it
// with both hands is not the reward -- landing the edge is (owner, 2026-10-23).
user float rsvr_axe_edge = 0.5;

// Axe lock-on (one-handed throw). Hold the throw and the axe paints the monster your
// hand is pointing at; release into the throw and it goes there. The cone is how far
// off your aim a target may be to be picked up; a lock already made is kept out to
// a little over twice that before it drops, so it does not flicker as you wind up.
user float rsvr_axe_lock_cone = 16.0;
user float rsvr_axe_lock_range = 1200.0;
