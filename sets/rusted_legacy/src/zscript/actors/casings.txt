Class BulletCasing : Actor
{
	Default
	{
	Projectile;
	Radius 1;
	Height 1;
	Speed 5;
	-NOGRAVITY;
	BounceType "Doom";
	bouncesound "flak/frag";
	bouncecount 3;
	BounceFactor 0.5;
	WallBounceFactor 0.1;
	+RANDOMIZE;
	+BOUNCEONFLOORS;
	+USEBOUNCESTATE;
	+CLIENTSIDEONLY;
	Gravity 0.5;
	}
	
	states
	{
	Spawn:
		BCAS A 1;
		Loop;
	Bounce:
		BCAS A 1;
		TNT1 A 0 A_SetPitch (pitch+random(-50,50), SPF_INTERPOLATE);
		TNT1 A 0 A_SetRoll (roll+random(-50,50), SPF_INTERPOLATE);
		TNT1 A 0 A_SetAngle (angle+random(-35,35),SPF_INTERPOLATE);
		TNT1 A 0 A_playsound("weapons/bulletdrop", CHAN_AUTO);
		TNT1 A 0 A_ChangeVelocity(frandom(-1,1),frandom(-1,1),frandom(-1,1));
		Goto Spawn;
	Death:
		TNT1 A 0 A_SetPitch (0);
		TNT1 A 0 A_SetRoll (0);
		TNT1 A 0 A_SetAngle (angle+random(-35,35),SPF_INTERPOLATE);
		BCAS A 17;
		Goto Fadeout;
	Fadeout:
		BCAS A 1 A_Fadeout(0.05);
		Loop;
	}
}
