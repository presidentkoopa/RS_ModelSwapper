Class BDExplosionSmoke : Actor
{
	mixin optimizeHelper;
	
	Default
	{
	+NOBLOCKMAP
	+NOTELEPORT
	+DONTSPLASH
	+MISSILE
	+FORCEXYBILLBOARD
	+CLIENTSIDEONLY
	+NOINTERACTION
	+NOGRAVITY
	+DOOMBOUNCE
	+THRUACTORS
	BounceFactor 0.5;
	Radius 0;
	Height 0;
	Alpha 0.01;
	RenderStyle "Translucent";
	Scale 1.2;
	Speed 1;
	Mass 0;
	}
	
	States
	{
	Spawn:
		TNT1 A 0;
		TNT1 A 0 A_JumpIf(waterlevel > 1, "Stop");
		TNT1 A 0 A_JumpIf(invoker.performanceMode, "DieFast");
		SMK2 AAAAAA 2 A_FadeIn(0.05);
		SMK2 AAAAAAAAAA 2;
		SMK2 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 2 A_FadeOut(0.01);
		Goto Death;
	DieFast:
		SMK2 AAAAA 1 A_FadeIn(0.05);
		SMK2 AAAAAAAAAAAAAAAAAAAAAAAAAAAAAA 1 A_FadeOut(0.01);
	Death:
		TNT1 A 0;
	Stop;	
		TNT1 A 0;
		Stop;
	}
}

{

	+FORCEXYBILLBOARD

	Height 1;
	Alpha 1.0;

	Speed 2;
	}

			A_JumpIf(waterlevel > 1, "Death");

	Death:
		TNT1 A 0;
		Stop;
	}
}
