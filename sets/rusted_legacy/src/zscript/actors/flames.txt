class BurningFXSpawner : Inventory
{
	transient int count;

	static const color partColors[] = 
	{
		"ffb37b",
		"f37317",
		"cb5707",
		"af4300"
	};	

	Default
	{
	Inventory.maxamount 1;
	}
	
	override void Doeffect()
	{
		if(!owner || !owner.bIsMonster)
		{
			destroy();
			return;
		}
		
		let fxSpawnPos = owner.pos + (0, 0, owner.height/2);
		Actor.Spawn("BurningFx", fxSpawnPos);
		if(GetAge() % 10 == 0) owner.A_StartSound("weapons/flame/death");
		
		if(owner.health < 1 && count < 70)
		{
		
			let lifetime = random[incin](30, 40);
			let spawnSize = random[incin](1, 3);
			let spawnVel = (0, 0, frandom[incin](0.8, 1));
			let accelz = (-spawnVel.z / lifetime);
			double vofs = 45 - (24 - frame*2); //the earlier in the animation, the lower the particles spawn.
			let spawnPos = level.Vec3Offset(owner.pos, (frandom(-8, 8), frandom(-8, 8), 15 + frandom(0, vofs)));
			let startalpha = alpha;
			let color1 = partColors[ random[incin](0, partColors.Size()-1) ];
			A_SpawnParticle(color1, SPF_FULLBRIGHT, lifetime, spawnSize, 0, spawnPos.x, spawnPos.y, spawnPos.z, spawnVel.x, spawnVel.y, spawnVel.z, accelz: accelz, startalphaf: startalpha);
			count++;
		}
		
		if(GetAge() > 35)
		{
			destroy();
			return;
		}
	}
}

class HellShotBurningStuff : Actor
{
	int lifespan;
	
	Default
	{
	Radius 12;
	Height 12;
	Speed 5;
	Scale 0.35;
	damage 0;
	damagetype "Fire";
	Renderstyle "Add";
	Gravity 0.2;
	Mass 0;
	PROJECTILE;
	+MISSILE
	+NOTELEPORT
	+BLOODLESSIMPACT 
	+THRUGHOST
	+NODAMAGETHRUST
	+NOGRAVITY
	+FLOAT
	+DONTFALL
	}
	
	override void BeginPlay()
	{
		lifespan = random(2,5) * 35;
	}
	
	override void Tick()
	{
		if(GetAge() && GetAge() > lifespan) SetStateLabel("Fadeout");
		Super.Tick();
	}
	
	States
	{
	Spawn:
		TNT1 A 0;
		TNT1 A 0 A_Jump(255, "Spawn1", "Spawn2", "Spawn3", "Spawn4", "live");
	Spawn1:
		TNT1 A 0 A_SetScale(scale.x * 0.9, scale.y * 1.3);
		Goto Death;
	Spawn2:
		TNT1 A 0 A_SetScale(scale.x * -0.9, scale.y * 0.8);
		Goto Death;
	Spawn3:
		TNT1 A 0 A_SetScale(scale.x * -1.1, scale.y * 0.9);
		Goto Death;
	Spawn4:
		TNT1 A 0 A_SetScale(scale.x * 1.2, scale.y * 1.1);
	Death:
		F1RE ABCDEFGH 2 Bright
		{
			if(waterlevel > 0) SetStateLabel("Remove");
			if(invoker.vel.x > 0 || invoker.vel.y > 0 || invoker.vel.z > 0)
			{
				invoker.destroy();
				return;
			}
			A_Fall();
		}
		TNT1 A 0 A_Explode(random(4,8), 32);
		Loop;
	Fadeout:
		F1RE ABCDEFGH 2 Bright
		{
			if(waterlevel > 0) SetStateLabel("Remove");
			A_ChangeVelocity(0, 0, 0);
			A_Fall();
			A_SetScale(scale.x * 0.9);
			A_Fadeout(0.05);
		}
		Loop;
	Remove:
		TNT1 A 0;
		Stop;
	}
}
