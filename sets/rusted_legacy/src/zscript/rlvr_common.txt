
mixin class optimizeHelper
{
	protected bool performanceMode;
	protected rlvr_Settings _settings;
	
	override void PostBeginPlay()
	{
		Super.PostBeginPlay();
			
		_settings = rlvr_Settings.of();
		performanceMode = _settings.perfMode();
	}
	
	protected double GetPitchTo(Actor source, Actor target, double zOfs = 0, double targZOfs = 0, bool absolute = false)
	{
		Vector3 origin = (source.pos.xy, source.pos.z - source.floorClip + zOfs);
		Vector3 dest = (target.pos.xy, target.pos.z - target.floorClip + targZOfs);

		Vector3 diff;
		if (!absolute)
			diff = level.Vec3Diff(origin, dest);
		else
			diff = dest - origin;

		return -atan2(diff.z, diff.xy.Length());
	}
}

Class RLVRHUDHandler : Eventhandler
{
	override void WorldTick()
	{
		if(_isTitlemap) return;
		
		Playerinfo player = Players[ConsolePlayer];
		if(!player) return;
		
		if(Level.Time % 100 == 0)
		{
			bool simpleHud = CVar.GetCVar("rl_simplehud", player).GetBool();
			
			if(simpleHud)
				player.mo.setinventory("vanillahud", 0);
			else
				player.mo.setinventory("vanillahud", 1);
		}
	}

	override void WorldLoaded(WorldEvent event)
	{
		_isTitlemap = CheckTitlemap();
	}

	private static bool CheckTitlemap()
	{
		bool isTitlemap = (level.mapname == "TITLEMAP");
		return isTitlemap;
	}
	
	private bool _isTitlemap;
}

//it loads the settings into player's inventory so other actors can check them without scanning for actual cvars
Class RLVRConfigHandler : Eventhandler
{
	override void WorldTick()
	{
		if(level.mapname == "TITLEMAP") return;
	
		Playerinfo player = Players[ConsolePlayer];
		if(!player) return;
		let pmo = player.mo;
		if(pmo.health < 1) return;
		
		if(level.time % 70 == 0) 
		{
			_settings 			= rlvr_Settings.of();
			int pickupModel 		= _settings.pickupModel();
			bool distModel 			= _settings.distModel();
			int modelDist 			= _settings.modelDist();
			
			if(!pmo.countinv("RLVRConfigItem")) pmo.giveinventory("RLVRConfigItem", 1);
			
			let configItem = RLVRConfigItem(pmo.findinventory("RLVRConfigItem"));
			
			configItem.pickupModel = pickupModel;
			configItem.distModel = distModel;
			configItem.modelDist = modelDist;
			
			if(developer == 2)
			{
				//auto-cycle between models for dev test
				if(test_count == 2)
					test_count = 0;
				else
					test_count++;
				configItem.pickupModel = test_count;
			}
		}
	}
	
	private rlvr_Settings _settings;
	private int test_count;
}

Class RLVRHapticHelper : EventHandler
{
	static void BuzzHand(int hand = 0)
	{
		Let player = Players[ConsolePlayer].mo;
		if(!player || player.health < 1) return;
		
		player.LineAttack(0, 0, 0, 1, "melee", "RLVRHapticDummy", LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ | (hand ? LAF_ISOFFHAND : 0));
	}
	
	static void Buzz()
	{
		RLVRHapticHelper.BuzzHand();
		RLVRHapticHelper.BuzzHand(1);
	}
	
	static void BuzzOffhand()
	{
		RLVRHapticHelper.BuzzHand(1);
	}
	
	static void BuzzWeapon(Weapon weap)
	{
		if(!weap || !weap.owner) return;
		
		int hand = weap.bOffhandWeapon ? 1 : 0;
		RLVRHapticHelper.BuzzHand(hand);
	}
}

class RLVRTipsHelper : EventHandler
{
	Array<string> missingKeys;
	int messageDelay;
	bool messagetDone;
	bool alertDone;
	bool hasMissingBinds;
	
	override void WorldTick()
	{
		if(level.mapname == "TITLEMAP") return;
		
		if(alertDone) return;
		
		PlayerInfo player = players[consoleplayer];
		bool showAlert = CVar.GetCvar("rl_show_tips", player).GetBool();
		
		if(!showAlert) return;
		
		//show queued missing keybind alert messages
		if(missingKeys.size() && !messageDelay)
		{
			let keyName = missingKeys[0];
			RLVR_HudMessage.Create(String.Format(Stringtable.Localize("$RLVRALERTMISSINGBIND"), keyName),
					fontname: "smallfont",
					fontColor: Font.CR_Orange,
					alignment: RLVR_HudMessage.ALIGN_CENTER,
					fadeInTime: 4,
					typetime: 1,
					holdTime: 70,
					fadeOutTime: 8
			);
			missingKeys.delete(0);
			messageDelay = 190;
		}
		
		if(hasMissingBinds && missingKeys.size() == 0 && !messageDelay)
		{
			alertDone = true;
			RLVR_HudMessage.Create(Stringtable.Localize("$RLVRALERTDONEMESSAGE"),
					fontname: "smallfont",
					fontColor: Font.CR_White,
					alignment: RLVR_HudMessage.ALIGN_CENTER,
					fadeInTime: 4,
					typetime: 1,
					holdTime: 100,
					fadeOutTime: 8
			);
		}
		
		if(messageDelay) messageDelay--;
	}
	
	override void PlayerEntered (PlayerEvent e)
	{
		if(level.mapname == "TITLEMAP") return;
		
		PlayerInfo player = players[e.PlayerNumber];
		bool showAlert = CVar.GetCvar("rl_show_tips", player).GetBool();
		
		//clear previously queued messages if exist
		let handler = RLVR_HudMessageHandler(StaticEventHandler.Find("RLVR_HudMessageHandler"));
		for (int i = 0; i < handler.hudmessages.Size(); i++)
		{
			let othermsg = handler.hudmessages[i];
			if (othermsg) othermsg.destroy();
		}
		
		if(showAlert)
		{
			if(missingKeys.size()) missingKeys.clear();
			Array<int> binds;
			
			Bindings.GetAllKeysForCommand(binds, "+reload");
			if (!binds.Size())
			{
				missingKeys.push("Reload");
				hasMissingBinds = true;
			}
			
			Bindings.GetAllKeysForCommand(binds, "+zoom");
			if (!binds.Size())
			{
				missingKeys.push("Zoom");
				hasMissingBinds = true;
			}
			
			Bindings.GetAllKeysForCommand(binds, "+oh_attack");
			if (!binds.Size())
			{
				missingKeys.push("Offhand Attack");
				hasMissingBinds = true;
			}
			
			Bindings.GetAllKeysForCommand(binds, "+oh_altatk");
			if (!binds.Size())
			{
				missingKeys.push("Offhand Alt-Attack");
				hasMissingBinds = true;
			}
		}
	}
}

Class RLVRMapCheck : EventHandler
{
	static bool isLoRMap()
	{
		let mapChecksum = Level.GetChecksum();
		
		bool match = false;
		for(int i = 0; i < RLVRMapCheck.LoR_Level_checksum.size(); i++)
		{
			if(mapChecksum == RLVRMapCheck.LoR_Level_checksum[i])
			{
				match = true;
				break;
			}
		}
		
		return match;
	}
	
	static const string LoR_Level_checksum[] = {
		"bf34c34c5dfc8bb47228cc304f9a6748", //map01 - Scar Gate
