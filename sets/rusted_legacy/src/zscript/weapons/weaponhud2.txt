	protected void setWeaponHUD(playerinfo player)
	{
		let pspr = player.GetPSprite(PSP_HHUD_BG);
		pspr.Caller = self;
		pspr.SetState(ResolveState("whud_bg"));
		
		pspr = player.GetPSprite(PSP_HHUD_MagDigit1);
		pspr.Caller = self;
		pspr.SetState(ResolveState("whud_mag_d1"));
		
		pspr = player.GetPSprite(PSP_HHUD_MagDigit2);
		pspr.Caller = self;
		pspr.SetState(ResolveState("whud_mag_d2"));
		
		pspr = player.GetPSprite(PSP_HHUD_MagDigit3);
		pspr.Caller = self;
		pspr.SetState(ResolveState("whud_mag_d3"));
		
		pspr = player.GetPSprite(PSP_HHUD_AmmoDigit1);
		pspr.Caller = self;
		pspr.SetState(ResolveState("whud_mag_d4"));
		
		pspr = player.GetPSprite(PSP_HHUD_AmmoDigit2);
		pspr.Caller = self;
		pspr.SetState(ResolveState("whud_mag_d5"));
		
		pspr = player.GetPSprite(PSP_HHUD_AmmoDigit3);
		pspr.Caller = self;
		pspr.SetState(ResolveState("whud_mag_d6"));
	}
	
	protected void clearWeaponHUD(playerinfo player)
	{
		let pspr = player.FindPSprite(PSP_HHUD_BG);
		if(pspr) pspr.Destroy();
		pspr = player.FindPSprite(PSP_HHUD_MagDigit1);
		if(pspr) pspr.Destroy();
		pspr = player.FindPSprite(PSP_HHUD_MagDigit2);
		if(pspr) pspr.Destroy();
		pspr = player.FindPSprite(PSP_HHUD_MagDigit3);
		if(pspr) pspr.Destroy();
		pspr = player.FindPSprite(PSP_HHUD_AmmoDigit1);
		if(pspr) pspr.Destroy();
		pspr = player.FindPSprite(PSP_HHUD_AmmoDigit2);
		if(pspr) pspr.Destroy();
		pspr = player.FindPSprite(PSP_HHUD_AmmoDigit3);
		if(pspr) pspr.Destroy();
	}
