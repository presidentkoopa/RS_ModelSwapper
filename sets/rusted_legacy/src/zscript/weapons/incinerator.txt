	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self)
		{
			checkOffhandGrab();
		}
	}
	
	void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		
		_puff = pmo.LineAttack(angle, 9, pitch-15, 0, "melee", "RLVRDUmmyPuff1", LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ | (hand ? LAF_ISOFFHAND : 0));
		vector3 gripPos1 = _puff.pos;
		
		double handDistGrip = level.Vec3Diff(gripPos1, owner.OffhandPos).Length();
		
		if(handDistGrip <= 4)
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}

	action void A_CheckImmersiveReloadAction1()
	{
		// RSRL (RSQ Weapon Depot): measured hand reload, row 12; his touch test below runs outside VR or with rsrl_handreload 0
		if (RSRL_Reload.On(invoker)) { bool rsDone; Vector3 rsAt; [rsDone, rsAt] = RSRL_Reload.Gate(invoker, 12, !invoker.ImmersiveAmmoGrab || invoker.owner.CountInv("grabbedAmmo") > 0); invoker.showWeaponMarker(rsAt); if (rsDone) invoker.setReloadStep(2); return; }
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle+10, 16, pitch-20, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 valvePos = _puff.pos;
		
		invoker.showWeaponMarker(valvePos);
		
		double handDistMag = level.Vec3Diff(valvePos, player.mo.OffhandPos).Length();
		if(handDistMag <= 3) invoker.setReloadStep(2);
	}

	action void A_ReloadCan()
	{
		let pmo = invoker.owner;
		let canammo = pmo.CountInv("FuelCan");
		let totalammo = pmo.CountInv("Cell");
		int addammo = 50 - canammo;
		addammo = addammo > totalammo ? totalammo : addammo;
		canammo += addammo;
		totalammo -= addammo;
		totalammo = totalammo < 0 ? 0 : totalammo;
		pmo.SetInventory("FuelCan", canammo);
		if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Cell", totalammo);
		invoker.weaponUnloaded = false;
	}
	
	action void A_UnloadCan()
	{
		let pmo = invoker.owner;
		int maxammo = RLVRMapCheck.isLoRMap() ? 150 : 300;
		if(pmo.CountInv("backpack")) maxammo *= 2;
		let canammo = pmo.CountInv("FuelCan");
		let totalammo = pmo.CountInv("Cell");
		if(totalammo + canammo > maxammo) return;
		pmo.SetInventory("FuelCan", 0);
		pmo.SetInventory("Cell", totalammo + canammo);
		invoker.weaponUnloaded = true;
	}

class RLIncineratorFlame : Actor
{
	Vector2 defscale;

	static const color partColors[] = 
	{
		"ffb37b",
		"f37317",
		"cb5707",
		"af4300"
	};	

	Default
	{
		Tag "Incinerator Flame";
		Projectile;
		Radius 13;
		Height 8;
		Damage 5;
		Speed 40;
		ExplosionDamage 5;
		ExplosionRadius 64;
		+FORCERADIUSDMG
		+BRIGHT
		RenderStyle 'Add';
		Alpha 0.75;
	}

	// Randomly mirror the sprite and randomize
	// its scale a bit to add more variety:
	override void PostBeginPlay()
	{
		Super.PostBeginPlay();
		bSPRITEFLIP = random[incin](0,1);
		scale *= frandom[incin](0.7, 1);
		defscale = scale; //record scale after randomization
	}

	override void Tick()
	{
		Super.Tick();
		// When the projectile dies, it sets its scale to 33%
		// of what it was. During death we'll gradually bring it back to scale:
		if (!isFrozen() && !bMissile)
		{
			scale.x = clamp(scale.x += 0.05, 0, defscale.x);
			scale.y = clamp(scale.y += 0.05, 0, defscale.y);
			// And we'll also raise its sprite every 2 tics:
			if (GetAge() % 2 == 0)
			{
				spriteoffset.y -= 1;
			}
			/*if (alpha >= default.alpha)
			{
				FSpawnParticleParams p;
				p.lifetime = random[incin](30, 40);
				p.size = random[incin](3, 6);
				p.style = STYLE_Add;
				p.flags = SPF_FULLBRIGHT;
				p.vel = (0, 0, frandom[incin](0.4, 0.6));
				p.accel.z = (-p.vel.z / p.lifetime);
				double vofs = 45 - (24 - frame*2); //the earlier in the animation, the lower the particles spawn.
				p.pos = level.Vec3Offset(pos, (frandom(-8, 8), frandom(-8, 8), 15 + frandom(0, vofs)));
				p.startalpha = alpha;
				p.fadestep = -1;
				p.color1 = partColors[ random[incin](0, partColors.Size()-1) ];
				level.SpawnParticle(p);
			}*/
			let lifetime = random[incin](30, 40);
			let spawnSize = random[incin](3, 6);
			let spawnVel = (0, 0, frandom[incin](0.4, 0.6));
			let accelz = (-spawnVel.z / lifetime);
			double vofs = 45 - (24 - frame*2); //the earlier in the animation, the lower the particles spawn.
			let spawnPos = level.Vec3Offset(pos, (frandom(-8, 8), frandom(-8, 8), 15 + frandom(0, vofs)));
			let startalpha = alpha;
			let color1 = partColors[ random[incin](0, partColors.Size()-1) ];

	States {
	Spawn:
		TNT1 A 2;
		IFLM A 1;
		IFLM B 2 A_StartSound("weapons/incinerator/burn");
		IFLM CDEFGH 2 A_FadeOut(0.1, FTF_Clamp);
		stop;
	Death:
		// The death sequence uses modified sprites made from the original
		// with the TEXTURES lump. Again, Arch-Vile-style flickering
		// between current and previous flame that the original sequence
		// had is rather hard on the eyes and not particularly pretty.
		// I added extra sprites from the Spawn sequence, and with added
		// scale and spriteoffset changes it appears smoother.
		TNT1 A 0 
		{
			A_StartSound("weapons/flame/death");
			scale *= 0.3;
			alpha = default.alpha;
		}
		IFLR A 3 A_Explode;
		IFLR B 3;
		IFLR C 2 A_Explode;
		IFLR D 4;
		IFLR E 2 A_Explode;
		IFLR F 4;
		IFLR G 2 A_StartSound("weapons/flame/fizzle");
		IFLR H 2 A_Explode();
		IFLR I 4;
		IFLR J 2 A_Explode();
		IFLR K 2;
		IFLR L 2 A_Explode();
		IFLR L 2
		{
			A_FadeOut(0.1);
		}
		wait;
	}
