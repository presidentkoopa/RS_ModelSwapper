
extend class DoomWeaponZ
{
	enum WLaserLayer
	{
        PSP_Laser_Pointer = 1070,
	}
	
	protected void setLaserPointer(playerinfo player)
	{
		let pspr = player.GetPSprite(PSP_Laser_Pointer);
		pspr.Caller = self;
		pspr.SetState(ResolveState("Laser"));
	}
	
	protected void clearLaserPointer(playerinfo player)
	{
		let pspr = player.FindPSprite(PSP_Laser_Pointer);
		if(pspr) pspr.Destroy();
	}
	
	States
	{
	Laser:
		TNT1 A 0 A_OverlayFlags(OverlayID(), PSPF_RENDERSTYLE | PSPF_FORCEALPHA, true);
		TNT1 A 0 A_OverlayRenderstyle(OverlayID(), STYLE_Add);
		TNT1 A 0 A_OverlayAlpha(OverlayID(), invoker.laserAlpha);
		TNT1 A 0 A_JumpIf(invoker.laserColor == 1, "Laser1");
		TNT1 A 0 A_JumpIf(invoker.laserColor == 2, "Laser2");
		TNT1 A 0 A_JumpIf(invoker.laserColor == 3, "Laser3");
		TNT1 A 0 A_JumpIf(invoker.laserColor == 4, "Laser4");
		TNT1 A 0 A_JumpIf(invoker.laserColor == 5, "Laser5");
		TNT1 A 0 A_JumpIf(invoker.laserColor == 6, "Laser6");
		TNT1 A 0 A_JumpIf(invoker.laserColor == 7, "Laser7");
	Laser0:
		LASR A 1 Bright;
		loop;
	Laser1:
		LASR B 1 Bright;
		loop;
	Laser2:
		LASR C 1 Bright;
		loop;
	Laser3:
		LASR D 1 Bright;
		loop;
	Laser4:
		LASR E 1 Bright;
		loop;
	Laser5:
		LASR F 1 Bright;
		loop;
	Laser6:
		LASR G 1 Bright;
		loop;
	Laser7:
		LASR H 1 Bright;
		loop;
	}
}
