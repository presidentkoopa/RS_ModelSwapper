	protected void calculateHUDValues(playerinfo player)
	{
		let pmo = player.mo;
	
		//Logics for ammo counter
		Ammo ammo1, ammo2;
		if(player.ReadyWeapon == self)
		{
			ammo1 = player.ReadyWeapon.Ammo1;
			ammo2 = player.ReadyWeapon.Ammo2;
		}
		else if(player.OffhandWeapon == self)
		{
			ammo1 = player.OffhandWeapon.Ammo1;
			ammo2 = player.OffhandWeapon.Ammo2;
		}
		
		if (ammo1 == NULL)
		{
			ammo1 = ammo2;
			ammo2 = NULL;
		}
		String ammocount2 = ammo1 != NULL ? ""..ammo1.Amount : "000";
		String ammocount1 = ammo2 != NULL ? ""..ammo2.Amount : "000";
	
		//Now set counter digits
		String clipAmmo, invAmmo;
		
		if(ammocount1.Length() == 1)
			clipAmmo = "00"..ammocount1;
		else if(ammocount1.Length() == 2)
			clipAmmo = "0"..ammocount1;
		else
			clipAmmo = ammocount1;
		
		if(ammocount2.Length() == 1)
			invAmmo = "00"..ammocount2;
		else if(ammocount2.Length() == 2)
			invAmmo = "0"..ammocount2;
		else
			invAmmo = ammocount2;
			
		let clipAmmoFirstChar = clipAmmo.Left(1);
		let clipAmmoSecondChar = clipAmmo.Mid(1,1);
		let clipAmmoThirdChar = clipAmmo.Mid(2,1);
			
		let invAmmoFirstChar = invAmmo.Left(1);
		let invAmmoSecondChar = invAmmo.Mid(1,1);
		let invAmmoThirdChar = invAmmo.Mid(2,1);
		
		clipCounter3 = clipAmmoFirstChar.toInt();
		clipCounter2 = clipAmmoSecondChar.toInt();
		clipCounter1 = clipAmmoThirdChar.toInt();
		
		ammoCounter3 = invAmmoFirstChar.toInt();
		ammoCounter2 = invAmmoSecondChar.toInt();
		ammoCounter1 = invAmmoThirdChar.toInt();
		
		//Logics for health and armor counter
		String pmoHealth = ""..pmo.health;
		String pmoArmor = ""..pmo.countinv("BasicArmor");
		
		String healthAmount, armorAmount;
		
		if(pmoHealth.Length() == 1)
			healthAmount = "00"..pmoHealth;
		else if(pmoHealth.Length() == 2)
			healthAmount = "0"..pmoHealth;
		else
			healthAmount = ""..pmoHealth;
		
		if(pmoArmor.Length() == 1)
			armorAmount = "00"..pmoArmor;
		else if(pmoArmor.Length() == 2)
			armorAmount = "0"..pmoArmor;
		else
			armorAmount = pmoArmor;
			
		let healthFirstChar = healthAmount.Left(1);
		let healthSecondChar = healthAmount.Mid(1,1);
		let healthThirdChar = healthAmount.Mid(2,1);
			
		let armorFirstChar = armorAmount.Left(1);
		let armorSecondChar = armorAmount.Mid(1,1);
		let armorThirdChar = armorAmount.Mid(2,1);
		
		hpCounter1 = healthFirstChar.toInt();
		hpCounter2 = healthSecondChar.toInt();
		hpCounter3 = healthThirdChar.toInt();
		
		armorCounter1 = armorFirstChar.toInt();
		armorCounter2 = armorSecondChar.toInt();
		armorCounter3 = armorThirdChar.toInt();
		
		if(wristWatchMode == 1)
		{
			watchCounter1 = clipCounter3;
			watchCounter2 = clipCounter2;
			watchCounter3 = clipCounter1;
			
			watchCounter4 = ammoCounter3;
			watchCounter5 = ammoCounter2;
			watchCounter6 = ammoCounter1;
		}
		else
		{
			watchCounter1 = hpCounter1;
			watchCounter2 = hpCounter2;
			watchCounter3 = hpCounter3;
			
			watchCounter4 = armorCounter1;
			watchCounter5 = armorCounter2;
			watchCounter6 = armorCounter3;
		}
		
		//logics for wrist mounted hud
		int pHealth, pArmor;
		pHealth = pmo.health;
		if(pHealth > 200) pHealth = 200;
		pArmor = pmo.countinv("BasicArmor");
		if(pArmor > 200) pArmor = 200;
		
		watchHP = (int)(ceil(pHealth / 5.0));
		watchArmor = (int)(ceil(pArmor / 5.0));
		
		pGod = pmo.bInvulnerable || (player.cheats & (CF_GODMODE | CF_GODMODE2));
		if(pGod && wristWatchMode == 0)
		{
			watchCounter1 = watchCounter2 = watchCounter3 = -1;
		}
		
		bool pInfiniteAmmo = player.cheats & CF_INFINITEAMMO;
		if(pInfiniteAmmo && wristWatchMode == 1)
		{
			watchCounter4 = watchCounter5 = watchCounter6 = -1;
		}
	}
