	Weapon.AmmoType2 "FuelCell";

	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self)
		{
			checkOffhandGrab();
			spawnEmbers();
		}
	}
	
	void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 7, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle, 11, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos2 = _puff.pos;
		
		double handDistGrip1 = level.Vec3Diff(gripPos1, owner.OffhandPos).Length();
		double handDistGrip2 = level.Vec3Diff(gripPos2, owner.OffhandPos).Length();
		
		if(handDistGrip1 <= 4 || handDistGrip2 <= 4 )
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}
	
	void spawnEmbers()
	{
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		let _puff = pmo.LineAttack(angle, 25, pitch+7, 0, "melee", "RLVRDUmmyPuff1", laflags);
		let spawnPos1 = _puff.pos;
		_puff = pmo.LineAttack(angle+17, 26, pitch+7, 0, "melee", "RLVRDUmmyPuff1", laflags);
		let spawnPos2 = _puff.pos;
		_puff = pmo.LineAttack(angle-17, 26, pitch+7, 0, "melee", "RLVRDUmmyPuff1", laflags);
		let spawnPos3 = _puff.pos;
		_puff = pmo.LineAttack(angle+25, 26, pitch+7, 0, "melee", "RLVRDUmmyPuff1", laflags);
		let spawnPos4 = _puff.pos;
		_puff = pmo.LineAttack(angle-25, 26, pitch+7, 0, "melee", "RLVRDUmmyPuff1", laflags);
		let spawnPos5 = _puff.pos;
		
		if(heatwaveFired)
		{
			Actor.Spawn("EmberSpawner3", spawnPos1);
			if(chargeBuiltUp == 3)
			{
				Actor.Spawn("EmberSpawner3", spawnPos2);
				Actor.Spawn("EmberSpawner3", spawnPos3);
			}
			if(chargeBuiltUp == 4)
			{
				Actor.Spawn("EmberSpawner3", spawnPos4);
				Actor.Spawn("EmberSpawner3", spawnPos5);
			}
			chargeBuiltUp = 0;
			heatwaveFired = false;
		}
		
		if(!heatwaveCharge) return;
		
		if(heatwaveCharge < 2)
		{
			Actor.Spawn("EmberSpawner2", spawnPos1);
		}
		if(heatwaveCharge < 3)
		{
			Actor.Spawn("EmberSpawner1", spawnPos1);
		}
		if(heatwaveCharge < 4)
		{
			Actor.Spawn("EmberSpawner1", spawnPos1);
			Actor.Spawn("EmberSpawner2", spawnPos2);
			Actor.Spawn("EmberSpawner2", spawnPos3);
		}
		if(heatwaveCharge == 4)
		{
			Actor.Spawn("EmberSpawner1", spawnPos2);
			Actor.Spawn("EmberSpawner1", spawnPos3);
			Actor.Spawn("EmberSpawner2", spawnPos4);
			Actor.Spawn("EmberSpawner2", spawnPos5);
		}
	}

	action void A_FireHeatWave()
	{
		A_StartSound("weapons/heatwave/fire", CHAN_WEAPON);
		A_Overlay(2001, "FlashEnd");

		double angle = 5;
		switch (invoker.heatwaveCharge)
		{
		case 2:
			angle = 12.5;
			break;
		case 3:
			angle = 20;
			break;
		case 4:
			angle = 27.5;
			break;
		case 5:
			angle = 35;
			break;
		}

		// In Legacy of Rust vertical autoaim is completely disabled for Heatwave,
		// presumably to prevent situations where a part of the "wave" is autoaimed
		// and a part isn't (supposedly ugly?)
		// Here instead we calculate the autoaim-affected slope with BulletSlope()
		// and then just unconditinally apply it to all projectiles, so if they
		// do get autoaimed, they'll be autoaimed together:
		double projPitch = BulletSlope();
		for (double ang = -angle; ang <= angle; ang += 5.0)
		{
			// Since autoaim is handled manually, we'll need FPF_NOAUTOAIM:
			A_FireProjectile("RLHeatWaveRipper", ang, useammo: false, flags: FPF_NOAUTOAIM, pitch: DeltaAngle(pitch, projPitch));
		}

		invoker.chargeBuiltUp = invoker.heatwaveCharge;
		takeinventory("FuelCell", invoker.heatwaveCharge);
		invoker.heatwaveCharge = 0;
		invoker.heatwaveFired = true;
		A_ClearRefire();
	}
	
	action void A_CheckImmersiveReloadAction1()
	{
		// RSRL (RSQ Weapon Depot): measured hand reload, row 19; his touch test below runs outside VR or with rsrl_handreload 0
		if (RSRL_Reload.On(invoker)) { bool rsDone; Vector3 rsAt; [rsDone, rsAt] = RSRL_Reload.Gate(invoker, 19, !invoker.ImmersiveAmmoGrab || invoker.owner.CountInv("grabbedAmmo") > 0); invoker.showWeaponMarker(rsAt); if (rsDone) invoker.setReloadStep(2, 1); return; }
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 11, pitch-25, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 magReloadPos = _puff.pos;
		
		invoker.showWeaponMarker(magReloadPos);
		
		double handDistMag = level.Vec3Diff(magReloadPos, player.mo.OffhandPos).Length();
		if(handDistMag <= 3) invoker.setReloadStep(2, 1);
	}

				Actor.Spawn("RLVRHeatFlash2", _puff.pos);
			else
				Actor.Spawn("RLVRHeatFlash", _puff.pos);
		}

	action void A_ReloadCells()
	{
		let pmo = invoker.owner;
		let magammo = pmo.CountInv("FuelCell");
		IF(magammo == 6) return;
		let totalammo = pmo.CountInv("Cell");
		int addammo;
		if(player.cheats & CF_INFINITEAMMO) totalammo = 600;
		if(totalammo >= 30)
		{
			int bfgcells = totalammo / 30;
			addammo = 6 - magammo;
			if(addammo > bfgcells) addammo = bfgcells;
			totalammo = totalammo - addammo * 30;
			pmo.SetInventory("FuelCell", magammo + addammo);
			if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Cell", totalammo);
			invoker.weaponUnloaded = false;
		}
	}
	
	action void A_UnloadMag()
	{
		let pmo = invoker.owner;
		int maxammo = RLVRMapCheck.isLoRMap() ? 150 : 300;
		if(pmo.CountInv("backpack")) maxammo *= 2;
		let magammo = pmo.CountInv("FuelCell");
		let totalammo = pmo.CountInv("Cell");
		int ejectammo = magammo * 30;
		if(totalammo + ejectammo > maxammo) return;
		pmo.SetInventory("FuelCell", 0);
		pmo.SetInventory("Cell", totalammo + ejectammo);
		invoker.weaponUnloaded = true;
	}
