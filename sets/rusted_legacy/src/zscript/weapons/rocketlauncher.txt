{
	transient int fireState;
	actor guidedMissile1, guidedMissile2, guidedMissile3;

	Default
	{
	Weapon.SelectionOrder 2501;
	Weapon.AmmoUse1 0;
	Weapon.AmmoGive1 3;
	Weapon.AmmoUse2 1;
	Weapon.AmmoGive2 0;
	Weapon.AmmoType1 "RocketAmmo";
	Weapon.AmmoType2 "RocketPack";
	+WEAPON.NOAUTOFIRE
	Inventory.PickupMessage "$GOTLAUNCHER";
	Tag "$TAG_ROCKETLAUNCHER";
	+WEAPON.NOAUTOAIM;
	+WEAPON.NOAUTOFIRE;
	+WEAPON.AMMO_OPTIONAL;
	+WEAPON.ALT_AMMO_OPTIONAL;
	+WEAPON.NOALERT;

	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self)
		{
			checkOffhandGrab();
			activateMissileGuidence();
		}
	}
	
	private void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 7, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle, 11, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos2 = _puff.pos;
		
		double handDistGrip1 = level.Vec3Diff(gripPos1, owner.OffhandPos).Length();
		double handDistGrip2 = level.Vec3Diff(gripPos2, owner.OffhandPos).Length();
		
		if(handDistGrip1 <= 4 || handDistGrip2 <= 4 )
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}
	
	private void activateMissileGuidence()
	{
		if(!guidedMissile1 && !guidedMissile2 && !guidedMissile3) return;
		
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		let _puff = pmo.LineAttack(angle, PLAYERMISSILERANGE, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 destPos = _puff.pos;

		if(guidedMissile1)
		{
			guidedMissile1.A_SetAngle(guidedMissile1.AngleTo(_puff));
			guidedMissile1.A_SetPitch(GetPitchTo(guidedMissile1, _puff));
			if(guidedMissile1.GetAge() > 35)
			{
				guidedMissile1.Vel3DFromAngle(50, guidedMissile1.angle, guidedMissile1.pitch);
			}
			else
			{
				guidedMissile1.SetOrigin(guidedMissile1.pos + (0,0,2), true);
				guidedMissile1.Vel3DFromAngle(GetDefaultSpeed("Rocket3"), guidedMissile1.angle, guidedMissile1.pitch);
			}
		}

		if(guidedMissile2)
		{
			guidedMissile2.A_SetAngle(guidedMissile2.AngleTo(_puff));
			guidedMissile2.A_SetPitch(GetPitchTo(guidedMissile2, _puff));
			if(guidedMissile2.GetAge() > 35)
			{
				guidedMissile2.Vel3DFromAngle(50, guidedMissile2.angle, guidedMissile2.pitch);
			}
			else
			{
				guidedMissile2.SetOrigin(guidedMissile2.pos + (0,0,2), true);
				guidedMissile2.Vel3DFromAngle(GetDefaultSpeed("Rocket3"), guidedMissile2.angle, guidedMissile2.pitch);
			}
		}

		if(guidedMissile3)
		{
			guidedMissile3.A_SetAngle(guidedMissile3.AngleTo(_puff));
			guidedMissile3.A_SetPitch(GetPitchTo(guidedMissile3, _puff));
			if(guidedMissile3.GetAge() > 35)
			{
				guidedMissile3.Vel3DFromAngle(50, guidedMissile3.angle, guidedMissile3.pitch);
			}
			else
			{
				guidedMissile3.SetOrigin(guidedMissile3.pos + (0,0,2), true);
				guidedMissile3.Vel3DFromAngle(GetDefaultSpeed("Rocket3"), guidedMissile3.angle, guidedMissile3.pitch);
			}
		}
	}
	
	action void A_CheckImmersiveReloadAction1()
	{
		// RSRL (RSQ Weapon Depot): measured hand reload, row 9; his touch test below runs outside VR or with rsrl_handreload 0
		if (RSRL_Reload.On(invoker)) { bool rsDone; Vector3 rsAt; [rsDone, rsAt] = RSRL_Reload.Gate(invoker, 9, !invoker.ImmersiveAmmoGrab || invoker.owner.CountInv("grabbedAmmo") > 0); invoker.showWeaponMarker(rsAt); if (rsDone) invoker.setReloadStep(2, 1); return; }
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle+75, 10, pitch+15, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 rocketReloadPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle+65, 6, pitch+35, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 rocketReloadPos2 = _puff.pos;
		
		invoker.showWeaponMarker(rocketReloadPos1);
		
		double handDistMag1 = level.Vec3Diff(rocketReloadPos1, invoker.owner.OffhandPos).Length();
		double handDistMag2 = level.Vec3Diff(rocketReloadPos2, invoker.owner.OffhandPos).Length();
		if(handDistMag1 <= 3 || handDistMag2 <= 3) invoker.setReloadStep(2, 1);
	}
	
	action void A_SpawnWeaponFlash()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 30, pitch+8, 0, "melee", "RLVRDUmmyPuff1", laflags);
		if(_puff) Actor.Spawn("RLVRRocketFlash", _puff.pos);
	}
	
	action void A_ReloadRockets()
	{
		let pmo = invoker.owner;
		let magammo = pmo.CountInv("RocketPack");
		let totalammo = pmo.CountInv("RocketAmmo");
		int addammo;
		if(magammo == 0)
		{
			addammo = 3;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		else
		{
			addammo = 3 - magammo;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		magammo += addammo;
		totalammo = totalammo - addammo;
		totalammo = totalammo < 0 ? 0 : totalammo;
		pmo.SetInventory("RocketPack", magammo);
		if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("RocketAmmo", totalammo);
		if(magammo == 3)
			invoker.fireState = 0;
		else if(magammo == 2)
			invoker.fireState = 1;
		else
			invoker.fireState = 2;
	}
