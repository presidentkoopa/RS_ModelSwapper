	Weapon.AmmoGive1 40;
	Weapon.AmmoUse2 1;
	Weapon.AmmoGive2 0;

	Weapon.AmmoType2 "PlasmaCell";

	+WEAPON.ALT_AMMO_OPTIONAL;
	+WEAPON.NOALERT;

	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self) checkOffhandGrab();
		
		//set the ammo display overlay
		if(owner.player.readyweapon is "RLPlasmaRifle" || owner.player.offhandweapon is "RLPlasmaRifle")
		{
			let pspr = owner.player.GetPSprite(2003);
			pspr.Caller = self;
			pspr.SetState(ResolveState("AmmoOverlay"));
			pspr = owner.player.GetPSprite(2004);
			pspr.Caller = self;
			pspr.SetState(ResolveState("AmmoOverlay2"));
			pspr = owner.player.GetPSprite(2005);
			pspr.Caller = self;
			pspr.SetState(ResolveState("AmmoOverlay3"));
		}
		else
		{
			let pspr = owner.player.FindPSprite(2003);
			if(pspr) pspr.Destroy();
			pspr = owner.player.FindPSprite(2004);
			if(pspr) pspr.Destroy();
			pspr = owner.player.FindPSprite(2005);
			if(pspr) pspr.Destroy();
		}
	}
	
	private void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 8, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos = _puff.pos;
		
		double handDistGrip = level.Vec3Diff(gripPos, owner.OffhandPos).Length();
		
		if(handDistGrip <= 4)
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}
	
	private void firePlasmaBall()
	{
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		let _puff = pmo.LineAttack(angle, 30, pitch+7, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 spawnPos = _puff.pos;
		
		_puff = pmo.LineAttack(angle, PLAYERMISSILERANGE, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 destPos = _puff.pos;
		
		let pball = pmo.SpawnPlayerMissile("NewPlasmaball");
		if(pball)
		{
			pball.SetOrigin(spawnPos, false);
			pball.A_SetAngle(pball.AngleTo(_puff));
			pball.A_SetPitch(GetPitchTo(pball, _puff));
			pball.Vel3DFromAngle(GetDefaultSpeed("NewPlasmaball"), pball.angle, pball.pitch);
			pmo.takeinventory("PlasmaCell", 1);
			
			if(!performanceMode)
			{
				int debrispawncount = random(4, 8);
				for(int i = 0; i < debrispawncount; i++)
				{
					let pdebris = Actor.Spawn("BDBluePlasmaPiece2", spawnPos);
					pdebris.Vel3DFromAngle(pdebris.default.speed, pdebris.AngleTo(_puff) + random(15, 40) * random(-1,1), random(-40,40));
				}
			}
			A_AlertMonsters();
		}
	}
	
	private void firePlasmaBlast()
	{
		let pmo = owner.player.mo;
		let totalammo = pmo.CountInv("PlasmaCell");
		int spentAmmo;
		
		if(beamCount < 3)
		{
			spentAmmo += 10;
			pmo.SpawnPlayerMissile("PlasmaBeamBlast", aimflags: ALF_NOFRIENDS);
			if(beamCount > 1 && totalammo >= 20)
			{
				pmo.SpawnPlayerMissile("PlasmaBeamBlast", aimflags: ALF_NOFRIENDS);
				spentAmmo += 10;
			}
		}
		else
		{
			if(totalammo >= 30)
			{
				pmo.SpawnPlayerMissile("PlasmaBeamBlast2", aimflags: ALF_NOFRIENDS);
				spentAmmo += 10;
			}
			if(beamCount > 3 && totalammo >= 42)
			{	
				pmo.SpawnPlayerMissile("PlasmaBeamBlast3", aimflags: ALF_NOFRIENDS);
				spentAmmo += 12;
			}
			if(beamCount == 5 && totalammo >= 55)
			{
				pmo.SpawnPlayerMissile("PlasmaBeamBlast4", aimflags: ALF_NOFRIENDS);
				spentAmmo += 13;
			}
		}
		pmo.A_StartSound("weapons/plasma/beamblast", CHAN_WEAPON);
		totalammo = totalammo - spentAmmo;
		if(!(owner.player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("PlasmaCell", totalammo);
		beamCount = 0;
	}
	
	private bool hasEnoughBeamAmmo()
	{
		let pmo = owner.player.mo;
		let totalammo = pmo.CountInv("PlasmaCell");
		int ammocount = beamCount * 10;
		if(beamCount > 3) ammocount += (beamCount - 3) * 2;
		if(beamCount > 4) ammocount += 1;
		if(ammocount > totalammo)
			return false;
		else
			return true;
	}

	action void A_FirePlasmaBlast()

	action void A_CheckImmersiveReloadAction1()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 8, pitch-15, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 magReloadPos = _puff.pos;
		
		invoker.showWeaponMarker(magReloadPos);
		
		double handDistMag = level.Vec3Diff(magReloadPos, player.mo.OffhandPos).Length();
		if(handDistMag <= 3) invoker.setReloadStep(2, 1);
	}

		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;

		_puff = pmo.LineAttack(angle, 30, pitch+7, 0, "melee", "RLVRDUmmyPuff1", laflags);
		if(_puff)
		{
			if(random[flash](0,9) > 5)
				Actor.Spawn("RLVRPlasmaFlash", _puff.pos);
			else
				Actor.Spawn("RLVRPlasmaFlash2", _puff.pos);
		}
	}
	
	action void A_ReloadMag()
	{
		let pmo = invoker.owner;
		let magammo = pmo.CountInv("PlasmaCell");
		let totalammo = pmo.CountInv("Cell");
		int addammo;
		if(magammo == 0)
		{
			addammo = 100;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		else
		{
			addammo = 100 - magammo;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		magammo += addammo;
		totalammo = totalammo - addammo;
		totalammo = totalammo < 0 ? 0 : totalammo;
		pmo.SetInventory("PlasmaCell", magammo);
		if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Cell", totalammo);
		invoker.weaponUnloaded = false;
	}
	
	action void A_UnloadMag()
	{
		let pmo = invoker.owner;
		int maxammo = RLVRMapCheck.isLoRMap() ? 150 : 300;
		if(pmo.CountInv("backpack")) maxammo *= 2;
		let magammo = pmo.CountInv("PlasmaCell");
		let totalammo = pmo.CountInv("Cell");
		if(totalammo + magammo > maxammo) return;
		pmo.SetInventory("PlasmaCell", 0);
		pmo.SetInventory("Cell", totalammo + magammo);
		invoker.weaponUnloaded = true;
	}
