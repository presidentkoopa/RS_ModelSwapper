	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self)
		{
			checkOffhandGrab();
			repositionEffectParticle();
		}
		
		//set the charge display overlay
		if((owner.player.readyweapon is "RLBFG9000" || owner.player.offhandweapon is "RLBFG9000") && charging)
		{
			if(GetAge() % 3 == 0)
			{
				if(charged < 7)
					charged++;
				else
					charged = 6;
			}
		
			let pspr = owner.player.GetPSprite(2006);
			pspr.Caller = self;
			pspr.SetState(ResolveState("ChargeOverlay"));
		}
		else
		{
			let pspr = owner.player.FindPSprite(2006);
			if(pspr) pspr.Destroy();
			charged = 0;
		}
	}
	
	private void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 11, pitch-20, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle, 10, pitch+20, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos2 = _puff.pos;
		
		double handDistGrip1 = level.Vec3Diff(gripPos1, owner.OffhandPos).Length();
		double handDistGrip2 = level.Vec3Diff(gripPos2, owner.OffhandPos).Length();
		
		if(handDistGrip1 <= 4 || handDistGrip2 <= 4)
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}

	action void A_ReloadPod()
	{
		let pmo = invoker.owner;
		let magammo = pmo.CountInv("BFGCell");
		let totalammo = pmo.CountInv("Cell");
		int addammo;
		if(player.cheats & CF_INFINITEAMMO) totalammo = 600;
		if(totalammo >= 40)
		{
			int bfgcells = totalammo / 40;
			addammo = 4 - magammo;
			if(addammo > bfgcells) addammo = bfgcells;
			totalammo = totalammo - addammo * 40;
			pmo.SetInventory("BFGCell", magammo + addammo);
			if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Cell", totalammo);
			invoker.weaponUnloaded = false;
		}
	}
	
	action void A_UnloadPod()
	{
		let pmo = invoker.owner;
		int maxammo = RLVRMapCheck.isLoRMap() ? 150 : 300;
		if(pmo.CountInv("backpack")) maxammo *= 2;
		let magammo = pmo.CountInv("BFGCell");
		let totalammo = pmo.CountInv("Cell");
		int ejectammo = magammo * 40;
		if(totalammo + ejectammo > maxammo) return;
		pmo.SetInventory("BFGCell", 0);
		pmo.SetInventory("Cell", totalammo + ejectammo);
		invoker.weaponUnloaded = true;
	}

class MiniBlackHole : PlasmaBall
{
	mixin optimizeHelper;

	Default
	{
	Speed 30;
	radius 20;
	height 40;
	Damage 50;
	SeeSound "";
	DeathSound "";
	+ROLLSPRITE
	+RIPPER
	//+PAINLESS
	+BLOODLESSIMPACT
	}
	
	private void pullNearbyMonsters()
	{
		int pullRadius = 512;
		BlockThingsIterator it = BlockThingsIterator.Create(self, pullRadius);
		Actor mo;

		while (it.Next())
		{
			mo = it.thing;
			if (mo && mo != self && Distance2D(mo) <= pullRadius && (mo.bMissile || (mo.bSolid && mo.bIsMonster)))
			{
				mo.giveinventory("GravityPullMonsters", 1);
				GravityPullMonsters(mo.findinventory("GravityPullMonsters")).pullSource = self;
			}
		}
	}
	
	States
	{
 	Spawn:
		TNT1 A 1 NoDelay
		{
			let outerCorona = Actor.Spawn("MiniBlackHoleOuterCorona");
			outerCorona.master = self;
			let corona = Actor.Spawn("MiniBlackHoleCorona");
			corona.master = self;
			let eye = Actor.Spawn("MiniBlackHoleEye");
			eye.master = self;
			A_StartSound("blackhole/active", CHAN_6);
		}
		TNT1 A 1 Bright
		{
			invoker.pullNearbyMonsters();
			if(random[blackhole](0,10) > 3 && !invoker.performanceMode)
			{
				vector3 randomSpawnPos = Vec3Angle(random(100,400), random(-90,90));
				let debris = Actor.Spawn("BHDebris", randomSpawnPos);
				debris.master = self;
			}
		}
		wait;
	Death:
		TNT1 AAAAAAAAAAAAAAAAAAA 10 Bright
		{
			invoker.pullNearbyMonsters();
			//A_Explode(20, 512, 0);
		}
		TNT1 AAAAAAAAAA 10 Bright
		{
			invoker.pullNearbyMonsters();
			//A_Explode(30, 512, 0);
			A_Jump(90, "Blast");
		}
		TNT1 AAAAAA 10 Bright
		{
			invoker.pullNearbyMonsters();
			//A_Explode(40, 512, 0);
			A_Jump(180, "Blast");
		}
	Blast:
		TNT1 A 1 
		{
			A_StopSOund(CHAN_6);
			A_SpawnItemEx("BlackHoleExplosion2",0,0,height*2/3);
			A_SpawnItemEx("BlackHoleExplosion",0,0,height*2/3);
			A_SpawnItemEx("BHExplosionParticles",0,0,height*2/3);
			A_Explode(200, 512, 0);
		}
		TNT1 A 2 A_StartSound("blackhole/explode");
		Stop;
	}
}

class GravityPullMonsters : Inventory
{
	mixin optimizeHelper;

	actor pullSource;
	bool init;
	int waitRelease;
	int pullSpeed;

	Default
	{
	Inventory.Maxamount 1;
	}
	
	override void DoEffect()
	{
		if(!owner || (owner && (!owner.bIsMonster || owner.bBoss || owner.bFriendly)))
		{
			destroy();
			return;
		}
	
		if(owner && pullSource && owner.health > 0)
		{
			let destpos = pullSource.vec3angle(random[blackhole](30,80), random[blackhole](-1,1)*random[blackhole](10,90));
			let diff = Level.Vec3Diff(owner.pos, destpos);
			let dir = diff.Unit();
			let dist = diff.Length();
		
			if(!init)
			{
				owner.bROLLSPRITE = true;
				owner.bNOGRAVITY = true;
				owner.A_SetRoll(owner.roll+random[blackhole](-1,1)*random[blackhole](5,10));
				owner.A_SetAngle(owner.AngleTo(pullSource));
				//owner.A_ChangeLinkFlags(true);
				pullSpeed = random[blackhole](10,25);
				init = true;
			}
			
			if(owner.health > 0 && !owner.InStateSequence(owner.curState, owner.ResolveState("Pain"))) owner.SetStateLabel("Pain");
			if(owner) owner.tics = -1;
			vector3 movePos;
			if(dist <= 25)
			{
				if(owner)
				{
					owner.A_ChangeLinkFlags(false);
					owner.A_SetRoll(owner.roll+15);
					movePos = destpos;
				}
			}
			else
			{
				//owner.A_SetRoll(owner.roll+4);
				if(owner) movePos = owner.pos + dir * 8;
			}
			if(owner && CheckMove(movePos.xy)) owner.SetOrigin(movePos, true);
			if(random[blackhole](0,10) > 8 && !performanceMode) spawnRippingParticles();
			if(owner && GetAge() && GetAge() % 10 == 0) owner.DamageMobj(pullSource, pullSource.target, random[blackhole](8,15), 'Fire', DMG_NO_PAIN | DMG_EXPLOSION, owner.AngleTo(pullSource));
			//owner.Vel3DFromAngle(pullSpeed, owner.AngleTo(pullSource)+45, GetPitchTo(owner, pullSource)+30);
		}
		else
		{
			if(owner.health < 1)
			{
				if(pullSource && !performanceMode)
				{
					owner.giveinventory("BodyRipFx", 1);
					BodyRipFx(owner.findinventory("BodyRipFx")).pullSource = pullSource;
				}
				release();
				return;
			}
		
			if(waitRelease == 0)
			{
				owner.A_SetRoll(0);
				owner.bROLLSPRITE = owner.default.bROLLSPRITE;
				owner.bNOGRAVITY = owner.default.bNOGRAVITY;
				owner.tics = 1;
				if(owner.FindState("Pain"))
					owner.SetStateLabel("Pain");
				else
					owner.SetStateLabel("Spawn");
				owner.A_ChangeLinkFlags(true);
				owner.Vel3DFromAngle(random(20,50), random(0,360), random(3,10));
			}
			
			if(waitRelease > 35)
			{
				owner.A_ChangeLinkFlags(false);
				destroy();
			}
			else
				waitRelease++;
			return;
		}
	
		Super.DoEffect();
	}
	
	protected void spawnRippingParticles()
	{
		if(owner)
		{
			vector3 spawnPos = owner.Vec3Angle(random(3,5),random(0,360)) + (0,0,owner.height*random(1,5)/5);
			let ripParticle = Actor.Spawn("BlackHoleRipParticle", spawnPos);
			ripParticle.Vel3DFromAngle(random[blackhole](2,5), owner.AngleTo(pullSource), GetPitchTo(owner, pullSource));
		}
	}
	
	private void release()
	{
		owner.bROLLSPRITE = owner.default.bROLLSPRITE;
		owner.bNOGRAVITY = owner.default.bNOGRAVITY;
		owner.tics = 1;
		owner.A_SetRoll(0);
		if(owner.health < 1) owner.A_Fall();
		if(owner.health > 0)
			owner.A_ChangeLinkFlags(false);
		else
			owner.A_ChangeLinkFlags(true);
		destroy();
	}
}
