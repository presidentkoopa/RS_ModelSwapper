
class RLShotgun : DoomWeaponZ replaces Shotgun
{
	Default
	{
	Weapon.SelectionOrder 1301;
	Weapon.AmmoUse1 0;
	Weapon.AmmoGive1 6;
	Weapon.AmmoUse2 1;
	Weapon.AmmoGive2 2;
	Weapon.AmmoType1 "Shell";
	Weapon.AmmoType2 "ShottyShells";
	Inventory.PickupMessage "$GOTSHOTGUN";
	Obituary "$OB_MPSHOTGUN";
	Tag "$TAG_SHOTGUN";
	+WEAPON.NOAUTOAIM;
	+WEAPON.NOAUTOFIRE;
	+WEAPON.AMMO_OPTIONAL;
	+WEAPON.NOALERT;
	}
	
	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self) checkOffhandGrab();
	}
	
	void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 9, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle, 14, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos2 = _puff.pos;
		
		double handDistGrip1 = level.Vec3Diff(gripPos1, owner.OffhandPos).Length();
		double handDistGrip2 = level.Vec3Diff(gripPos2, owner.OffhandPos).Length();
		
		if(handDistGrip1 <= 4 || handDistGrip2 <= 4 )
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}
	
	action void A_CheckImmersiveReloadAction1()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 6, pitch-15, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 magReloadPos = _puff.pos;
		
		invoker.showWeaponMarker(magReloadPos);
		
		double handDistMag = level.Vec3Diff(magReloadPos, invoker.owner.OffhandPos).Length();
		if(handDistMag <= 2) invoker.setReloadStep(2, 1);
	}
	
	action void A_CheckImmersiveReloadAction2()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 9, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle, 14, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos2 = _puff.pos;
		
		invoker.showWeaponMarker(gripPos2);
		
		double handDistGrip1 = level.Vec3Diff(gripPos1, invoker.owner.OffhandPos).Length();
		double handDistGrip2 = level.Vec3Diff(gripPos2, invoker.owner.OffhandPos).Length();
		if(handDistGrip1 <= 2.5 || handDistGrip2 <= 2.5 ) invoker.setReloadStep(3);
	}
	
	action void A_SpawnWeaponFlash()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 27, pitch+5, 0, "melee", "RLVRDUmmyPuff1", laflags);
		if(_puff) Actor.Spawn("RLVRShotgunFlash", _puff.pos);
	}
	
	action void A_ReloadShells()
	{
		let pmo = invoker.owner;
		let shottyammo = pmo.CountInv("ShottyShells");
		let totalammo = pmo.CountInv("Shell");
		int addammo;
		if(shottyammo < 6)
		{
			addammo = 1;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		shottyammo += addammo;
		totalammo = totalammo - addammo;
		totalammo = totalammo < 0 ? 0 : totalammo;
		pmo.SetInventory("ShottyShells", shottyammo);
		if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Shell", totalammo);
	}
	
	action void A_UnloadShells()
	{
		let pmo = invoker.owner;
		int maxammo = pmo.CountInv("backpack") ? 100 : 50;
		let shottyammo = pmo.CountInv("ShottyShells");
		let totalammo = pmo.CountInv("Shell");
		if(totalammo + shottyammo > maxammo) return;
		If(!invoker.performanceMode && shottyammo)
		{
			for(int i=0; i<shottyammo; i++)
				A_FireProjectile("ShotgunShellSpawner", 0, 0, 0, 0, FPF_NOAUTOAIM, 0);
		}
		pmo.SetInventory("ShottyShells", 0);
		pmo.SetInventory("Shell", totalammo + shottyammo);
		
	}
	
	States
	{
	Ready:
		SHTG # 0 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Loop;
	Deselect:
		SHTG # 0 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### A 1 A_Lower (160);
		Loop;
	Select:
		SHTG # 0 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### A 1 A_Raise (160);
		Loop;
	DryFire:
		#### # 0 A_JumpIf(invoker.autoReload, "Reload");
		SHTG # 0 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### # 0 A_StartSound("weapon/dryfire", CHAN_WEAPON);
		#### B 10;
		Goto Ready;
	Fire:
		SHTG # 0 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### A 0 A_JumpIfInventory("ShottyShells", 1, 1);
		Goto DryFire;
		#### BB 1 A_AlertMonsters;
		#### # 0 A_StartSound("weapons/shotgun/fire", CHAN_WEAPON);
		#### # 0 A_FireBullets(5, 1, 4, 13, "EnhancedShotPuff", FBF_NORANDOM);
		#### # 0 A_FireBullets(2.5, 2.5, 5, 13, "EnhancedShotPuff", FBF_NORANDOM);
		#### # 0 A_JumpIf(invoker.disableTracer, 6);
		#### ##### 0 A_FireProjectile("Bullettracer", frandom(-3, 3), 0, 0, 0, FPF_NOAUTOAIM, 0);
		#### # 0 A_TakeInventory("ShottyShells", 1);
		#### C 1 A_GunFlash("MuzzleFlash");
		#### # 0 A_SpawnWeaponFlash;
		#### # 0 A_FireProjectile("ShotWeaponBarrelSmoke",0,0,0,0,FPF_NOAUTOAIM,0);
		SHTG DDEEFFGGGGGHH 1 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### # 0 A_StartSound("weapons/shotgun/pump1", CHAN_WEAPON, CHANF_OVERLAP);
		#### # 0 A_OffhandHaptic; 								//haptic
		SHTG IIJJ 1 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### # 0 A_JumpIf(invoker.performanceMode, 2);
		#### # 0 A_FireProjectile("ShotgunShellSpawner", 0, 0, 0, 0, FPF_NOAUTOAIM, 0);
		SHTG KKLLMM 1 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### # 0 A_StartSound("weapons/shotgun/pump2", CHAN_WEAPON, CHANF_OVERLAP);
		SHTG AAAAAAA 1 {
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_ReFire();
		}
		Goto Ready;
	Reload:
		#### # 0 A_JumpIf(invoker.owner.countinv("ShottyShells") > 5, "ReloadDone");
		#### # 0 A_JumpIfInventory("Shell", 1, 1);
		Goto ReloadNoAmmo;
		#### # 0 A_JumpIf(invoker.ImmersiveReload, "ImmersiveReload");
		SHTG NNNNNNNNOP 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_PlaySound("weapons/shotgun/reload", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		SHTG QQQQQQQQQ 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_ReloadShells;
		#### # 0 A_JumpIf((invoker.owner.countinv("ShottyShells") < 6 && invoker.owner.countinv("Shell")), "Reload");
	ReloadDone:
		#### # 0 { invoker.hideWeaponMarker(); }
		SHTG AAAAAAAAII 1 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### # 0 A_StartSound("weapons/shotgun/pump1", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		#### JJJJKKLLMM 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_StartSound("weapons/shotgun/pump2", CHAN_WEAPON);
		#### AA 1;
		Goto Ready;
	ReloadNoAmmo:
		#### ##### 1;
		Goto Ready;
	Zoom:
	UnloadAmmo:
		SHTG AAAAAAAAII 1 A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
		#### # 0 A_StartSound("weapons/shotgun/pump1", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		#### # 0 A_UnloadShells;
		#### JJJJKKLLMM 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_StartSound("weapons/shotgun/pump2", CHAN_WEAPON);
		Goto Ready;
	ImmersiveReload:
		#### # 0 { invoker.setReloadStep(1); }
		SHTG A 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_CheckImmersiveReloadAction1();
		#### # 0 A_JumpIf(player.mo.countinv("ImmersiveReloadStep") == 2, "ImmersiveReloadMag");
		Loop;
	ImmersiveReloadMag:
		#### # 0 { invoker.hideWeaponMarker(); }
		SHTG NNNOOOPPP 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_PlaySound("weapons/shotgun/reload", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		SHTG QQQQQ 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_ReloadShells;
		#### # 0 A_JumpIfInventory("Shell", 1, 1);
		Goto WaitReload;
		#### # 0 A_JumpIf(invoker.owner.countinv("ShottyShells") < 6, 2);
		#### # 0 { invoker.setReloadStep(0); }
		Goto WaitReload;
		SHTG NNNOOOPPP 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_PlaySound("weapons/shotgun/reload", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		SHTG QQQQQ 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_ReloadShells;
		//#### # 0 A_JumpIf((invoker.owner.countinv("ShottyShells") < 6 && invoker.owner.countinv("ShottyShells")), "ImmersiveReload");
		#### # 0 { invoker.setReloadStep(0); }
	WaitReload:
		SHTG A 1
		{
			A_SetAltModel("SHTP", "SH2H", "SH2P", "SHNH", "SHNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_CheckImmersiveReloadAction2();
		#### # 0 A_JumpIf(player.mo.countinv("ImmersiveReloadStep") == 3, "ReloadDone");
		Loop;
	MuzzleFlash:
		TNT1 A 0 A_OverlayFlags(OverlayID(), PSPF_RENDERSTYLE | PSPF_FORCEALPHA, true);
		TNT1 A 0 A_OverlayRenderstyle(OverlayID(), STYLE_Add);
		TNT1 A 0 A_OverlayAlpha(OverlayID(), invoker.flashAlpha);
		TNT1 A 0 A_JumpIf(invoker.showFLashSprite == false, 2);
		TNT1 A 0 A_Jump(256, "MuzzleFlash1", "MuzzleFlash2", "MuzzleFlash3", "MuzzleFlash4", "MuzzleFlash5");
		TNT1 A 0;
		stop;
	MuzzleFlash1:
		SHTF A 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash2:
		SHTF B 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash3:
		SHTF C 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash4:
		SHTF D 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash5:
		SHTF E 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	Spawn:
		SHOT A -1;
		Stop;
	PlaceHolder:
		SHTP A 1;
		SH2H A 1;
		SH2P A 1;
		SHNH A 1;
		SHNP A 1;
	}
