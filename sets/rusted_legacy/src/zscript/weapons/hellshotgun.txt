
class HellShotgun : DoomWeaponZ
{
	bool gunReloading;
	int immersiveLeverActionWait;

	Default
	{
	Weapon.SelectionOrder 1302;
	Weapon.AmmoUse1 0;
	Weapon.AmmoGive1 6;
	Weapon.AmmoUse2 1;
	Weapon.AmmoGive2 2;
	Weapon.AmmoType1 "Shell";
	Weapon.AmmoType2 "HellShells";
	Inventory.PickupMessage "$GOTHELLSHOTGUN";
	Obituary "$OB_MPHELLSHOTGUN";
	Tag "$TAG_HELLSHOTGUN";
	+WEAPON.NOAUTOAIM;
	+WEAPON.NOAUTOFIRE;
	+WEAPON.AMMO_OPTIONAL;
	+WEAPON.NOALERT;
	}
	
	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self) checkOffhandGrab();
		
		//set the demonic runes
		if(owner.player.readyweapon is "HellShotgun" || owner.player.offhandweapon is "HellShotgun")
		{
			let pspr = owner.player.FindPSprite(2010);
			if(!pspr && CheckWeaponState("Ready"))
			{
				pspr = owner.player.GetPSprite(2010);
				pspr.Caller = self;
				pspr.SetState(ResolveState("RunicOverlay"));
			}
			else
			{
				if(!CheckWeaponState("Ready"))
				{
					if(pspr) pspr.Destroy();
				}
			}
		}
		else
		{
			let pspr = owner.player.FindPSprite(2010);
			if(pspr) pspr.Destroy();
		}
	}
	
	void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 9, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle, 14, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos2 = _puff.pos;
		
		double handDistGrip1 = level.Vec3Diff(gripPos1, owner.OffhandPos).Length();
		double handDistGrip2 = level.Vec3Diff(gripPos2, owner.OffhandPos).Length();
		
		if(handDistGrip1 <= 4 || handDistGrip2 <= 4 )
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}
	
	action bool A_CheckImmersiveLevelAction1()
	{
		let pmo = player.mo;
		bool mainhandThrow, offhandThrow;
		
		bool ovrdrive_loaded = CVar.GetCVar("ovrdrive_loaded") != NULL;
		if(ovrdrive_loaded) {
		
			// ===== Using api from oVRdrive mod to detect motion throw =====
			// it has fail-safe so doesn't crash if oVRdrive is not loaded
			
			string throwhMotionClass1 = "vmMainhandThrow";
			string throwhMotionClass2 = "vmOffhandThrow";
			class<Inventory> mainhandThrowHandler = throwhMotionClass1;
			class<Inventory> offhandThrowHandler = throwhMotionClass2;

			if(invoker.bOffhandWeapon)
			{
				if(offhandThrowHandler && pmo.countInv(offhandThrowHandler))
					offhandThrow = true;
			}
			else
			{
				if(mainhandThrowHandler && pmo.countInv(mainhandThrowHandler))
					mainhandThrow = true;
			}
			
			// ===== End oVRdrive api =====
			
			return (mainhandThrow || offhandThrow);
		}
		else
		{
			return true; //if motion throw not available, fall-back to default lever action animation
		}
	}
	
	action void A_CheckImmersiveReloadAction1()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 8, pitch+45, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 magReloadPos = _puff.pos;
		
		invoker.showWeaponMarker(magReloadPos);
		
		double handDistMag = level.Vec3Diff(magReloadPos, invoker.owner.OffhandPos).Length();
		if(handDistMag <= 2) invoker.setReloadStep(2, 1);
	}
	
	action void A_CheckImmersiveReloadAction2()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = LineAttack(angle, 10, pitch-10, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos = _puff.pos;
		
		invoker.showWeaponMarker(gripPos);
		
		double handDistGrip = level.Vec3Diff(gripPos, invoker.owner.OffhandPos).Length();
		if(handDistGrip <= 2.5) invoker.setReloadStep(3);
	}
	
	action void A_SpawnWeaponFlash()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 27, pitch+5, 0, "melee", "RLVRDUmmyPuff1", laflags);
		if(_puff) Actor.Spawn("RLVRShotgunFlash", _puff.pos);
	}
	
	action void A_ReloadShells()
	{
		let pmo = invoker.owner;
		let shottyammo = pmo.CountInv("HellShells");
		let totalammo = pmo.CountInv("Shell");
		int addammo;
		if(shottyammo < 5 && totalammo >= 2)
		{
			addammo = 1;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		shottyammo += addammo;
		totalammo = totalammo - addammo * 2;
		totalammo = totalammo < 0 ? 0 : totalammo;
		pmo.SetInventory("HellShells", shottyammo);
		if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Shell", totalammo);
	}
	
	action void A_UnloadShells()
	{
		let pmo = invoker.owner;
		let shottyammo = pmo.CountInv("HellShells");
		let totalammo = pmo.CountInv("Shell");
		if(totalammo + shottyammo * 2 > 100) return;
		If(!invoker.performanceMode && shottyammo)
		{
			for(int i=0; i<shottyammo; i++)
				A_FireProjectile("HellShellCasingSpawner", 0, 0, 0, 0, FPF_NOAUTOAIM, 0);
		}
		pmo.SetInventory("HellShells", 0);
		pmo.SetInventory("Shell", totalammo + shottyammo * 2);
		
	}
	
	action bool CheckActiveMotionAction()
	{
		bool ovrdrive_loaded = CVar.GetCVar("ovrdrive_loaded") != NULL;
		return ovrdrive_loaded;
	}
	
	States
	{
	Ready:
		TNT1 A 0;
		HSHT # 0 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
		#### A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		#### A 0 { invoker.gunReloading = false; }
		Loop;
	Deselect:
		HSHT # 0 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
		#### A 1 A_Lower (160);
		Loop;
	Select:
		HSHT # 0 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
		#### A 1 A_Raise (160);
		Loop;
	DryFire:
		#### # 0 A_JumpIf(invoker.autoReload, "Reload");
		HSHT # 0 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
		#### # 0 A_StartSound("weapon/dryfire", CHAN_WEAPON);
		#### B 10;
		Goto Ready;
	Fire:
		#### A 0 A_JumpIfInventory("HellShells", 1, 1);
		Goto DryFire;
		HSHT # 0 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
		#### BB 1 A_AlertMonsters;
		#### # 0 A_StartSound("weapons/shotgun/fire", CHAN_WEAPON, volume: 0.7);
		#### # 0 A_StartSound("weapons/hellshot/fireburn", CHAN_WEAPON, CHANF_OVERLAP);
		#### ########## 0
		{
			let hellshot = A_FireProjectile("HellShotMissile", frandom(-4.5, 4.5), 0, 0, 3, FPF_NOAUTOAIM, 0);
			if(hellshot)
			{
				double pitchRandom = -(invoker.bOffhandWeapon ? player.mo.OffhandPitch : player.mo.AttackPitch) + frandom(-1.5, 1.5);
				hellshot.Vel3DFromAngle(hellshot.default.speed, hellshot.angle, pitchRandom);
			}
		}
		#### # 0 A_TakeInventory("HellShells", 1);
		HSHT # 0 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
		#### CC 1 A_GunFlash("MuzzleFlash");
		#### # 0 A_SpawnWeaponFlash;
		#### # 0 A_FireProjectile("ShotWeaponBarrelSmoke",0,0,0,0,FPF_NOAUTOAIM,0);
		HSHT DDEEFFGGGGG 1 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
	EjectShell:
		#### # 0 A_JumpIf(CheckActiveMotionAction(), "ImmersiveLeverAction");
	LeverAction:
		HSH2 AB 1 A_SetAltModel("H2SP", "H2H2", "H2P2", "HNH2", "HNP2");
		#### # 0 A_StartSound("weapons/hellshot/open", CHAN_WEAPON, CHANF_OVERLAP);
		#### # 0 A_JumpIf(invoker.performanceMode, 2);
		#### # 0 A_FireProjectile("HellShellCasingSpawner", 0, 0, 0, 0, FPF_NOAUTOAIM, 0);
		HSH2 C 1 A_SetAltModel("H2SP", "H2H2", "H2P2", "HNH2", "HNP2");
		HSH2 EFGHJKLMOPQT 1 A_SetAltModel("H2SP", "H2H2", "H2P2", "HNH2", "HNP2");
		#### # 0 A_StartSound("weapons/hellshot/close", CHAN_WEAPON, CHANF_OVERLAP);
		HSHT AAAAAAAA 1 A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
		Goto Ready;
	ImmersiveLeverAction:
		#### # 0 { invoker.immersiveLeverActionWait = 0; }
		#### # 0 A_JumpIf(A_CheckImmersiveLevelAction1(), "LeverAction");
		HSHT A 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
			invoker.immersiveLeverActionWait++;
			if(invoker.immersiveLeverActionWait >= 35) //if player fail to activate gesture it'll automatically show lever action animation after 1 sec
				return ResolveState("LeverAction");
			else
				return ResolveState(null);
		}
		Goto ImmersiveLeverAction+1;
	Reload:
		#### AAAAAAAA 1 A_JumpIf(invoker.gunReloading, "ImmersiveReload");
		#### AAAAAAAA 1 A_JumpIf(invoker.owner.countinv("HellShells") > 4, "EjectShell");
		#### # 0 A_JumpIfInventory("Shell", 2, 1);
		Goto ReloadNoAmmo;
		HSHT HIJ 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_StartSound("weapons/hellshot/open", CHAN_WEAPON);
		#### # 0 A_JumpIf(invoker.ImmersiveReload, "ImmersiveReload");
	InsertAmmo:
		HSHT KKKKKKLLMMNN 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_PlaySound("weapons/shotgun/reload", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		#### # 0 A_ReloadShells;
		//HSH2 OOOOOOOO 1 A_SetAltModel("H2SP", "H2H2", "H2P2", "HNH2", "HNP2");
		#### # 0 A_JumpIf((invoker.owner.countinv("HellShells") < 5 && invoker.owner.countinv("Shell") > 1), "InsertAmmo");
	ReloadDone:
		#### # 0
		{
			invoker.gunReloading = false;
			invoker.hideWeaponMarker();
		}
		HSHT PPQQRR 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_StartSound("weapons/hellshot/close", CHAN_WEAPON);
		#### AA 1;
		Goto Ready;
	ReloadNoAmmo:
		#### ##### 1;
		Goto Ready;
	Zoom:
	UnloadAmmo:
		#### # 0 A_JumpIfInventory("HellShells", 1, 1);
		Goto Ready;
		HSHT HIJ 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_StartSound("weapons/hellshot/open", CHAN_WEAPON);
		#### # 0 A_UnloadShells;
		HSHT PQR 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_StartSound("weapons/hellshot/close", CHAN_WEAPON);
		Goto Ready;
	ImmersiveReload:
		#### # 0
		{
			invoker.gunReloading = true;
			invoker.setReloadStep(1);
		}
		HSHT J 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_CheckImmersiveReloadAction1();
		#### # 0 A_JumpIf(player.mo.countinv("ImmersiveReloadStep") == 2, "ImmersiveReloadMag");
		Loop;
	ImmersiveReloadMag:
		HSHT KKLLMMNNNNN 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_PlaySound("weapons/shotgun/reload", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		#### # 0 A_ReloadShells;
		#### # 0 A_JumpIfInventory("Shell", 2, 1);
		Goto WaitReload;
		#### # 0 A_JumpIf(invoker.owner.countinv("HellShells") < 5, 2);
		#### # 0 { invoker.setReloadStep(0); }
		Goto WaitReload;
		HSHT KKLLMMNNNNN 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_PlaySound("weapons/shotgun/reload", CHAN_WEAPON);
		#### # 0 A_OffhandHaptic; 								//haptic
		#### # 0 A_ReloadShells;
		//#### # 0 A_JumpIf((invoker.owner.countinv("HellShells") < 5 && invoker.owner.countinv("Shell") > 1), "InsertAmmo");
		#### # 0 { invoker.setReloadStep(0); }
	WaitReload:
		HSHT J 1
		{
			A_SetAltModel("HSHP", "HS2H", "HS2P", "HSNH", "HSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_CheckImmersiveReloadAction2();
		#### # 0 A_JumpIf(player.mo.countinv("ImmersiveReloadStep") == 3, "ReloadDone");
		Loop;
	MuzzleFlash:
		TNT1 A 0 A_OverlayFlags(OverlayID(), PSPF_RENDERSTYLE | PSPF_FORCEALPHA, true);
		TNT1 A 0 A_OverlayRenderstyle(OverlayID(), STYLE_Add);
		TNT1 A 0 A_OverlayAlpha(OverlayID(), invoker.flashAlpha);
		TNT1 A 0 A_JumpIf(invoker.showFLashSprite == false, 2);
		TNT1 A 0 A_Jump(256, "MuzzleFlash1", "MuzzleFlash2", "MuzzleFlash3", "MuzzleFlash4", "MuzzleFlash5");
		TNT1 A 0;
		stop;
	MuzzleFlash1:
		SHTF A 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash2:
		SHTF B 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash3:
		SHTF C 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash4:
		SHTF D 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash5:
		SHTF E 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	RunicOverlay:
		HRUN BCDEFGHIJKLMNOPQRST 35 Bright;
		loop;
	Spawn:
		HELS A -1;
		Stop;
	PlaceHolder:
		HSH2 A 1;
		HSHP A 1;
		H2SP A 1;
		HS2H A 1;
		H2H2 A 1;
		HS2P A 1;
		H2P2 A 1;
		HSNH A 1;
		HNH2 A 1;
		HSNP A 1;
		HNP2 A 1;
	}
