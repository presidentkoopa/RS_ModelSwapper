	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self) checkOffhandGrab();
		
		//set the ammo belt overlay
		if(owner.player.readyweapon is "RLChaingun" || owner.player.offhandweapon is "RLChaingun")
		{
			let pspr = owner.player.GetPSprite(2002);
			pspr.Caller = self;
			if(!loadingAmmo) pspr.SetState(ResolveState("AmmoOverlay"));
		}
		else
		{
			let pspr = owner.player.FindPSprite(2002);
			if(pspr) pspr.Destroy();
		}
	}
	
	private void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 9.5, pitch+15, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos1 = _puff.pos;
		
		_puff = pmo.LineAttack(angle, 8, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 gripPos2 = _puff.pos;
		
		double handDistGrip1 = level.Vec3Diff(gripPos1, owner.OffhandPos).Length();
		double handDistGrip2 = level.Vec3Diff(gripPos2, owner.OffhandPos).Length();
		
		if(handDistGrip1 <= 4 || handDistGrip2 <= 4)
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}
	
	action void A_FireBullets2(double spread_xy, double spread_z, int numbullets, int damageperbullet, class<Actor> pufftype = "BulletPuff", int flags = 1, double range = 0, class<Actor> missile = null, double Spawnheight = 32, double Spawnofs_xy = 0)
	{
		let player = player;
		if (!player) return;

		let pawn = PlayerPawn(self);
		let weapon = player.ReadyWeapon;

		int i;
		double bangle;
		double bslope = 0.;
		int laflags = (flags & FBF_NORANDOMPUFFZ)? LAF_NORANDOMPUFFZ : 0;
		FTranslatedLineTarget t;
		actor puff;

		if ((flags & FBF_USEAMMO) && weapon &&  stateinfo != null && stateinfo.mStateType == STATE_Psprite)
		{
			if (!weapon.DepleteAmmo(weapon.bAltFire, true))
				return;	// out of ammo
		}
		
		if (range == 0)	range = PLAYERMISSILERANGE;

		if (!(flags & FBF_NOFLASH)) pawn.PlayAttacking2 ();

		if (!(flags & FBF_NOPITCH)) bslope = BulletSlope();
		bangle = Angle;

		if (pufftype == NULL) pufftype = 'BulletPuff';

		if (weapon != NULL)
		{
			A_StartSound(weapon.AttackSound, CHAN_WEAPON);
		}

		if (!player.refire || numbullets == 0)
		{
			int damage = damageperbullet;

			if (!(flags & FBF_NORANDOM))
				damage *= random[cabullet](1, 3);

			puff = LineAttack(bangle, range, bslope, damage, 'Hitscan', pufftype, laflags, t);
		}
		else 
		{
			if (numbullets < 0)
				numbullets = 1;
			for (i = 0; i < numbullets; i++)
			{
				double pangle = bangle;
				double slope = bslope;

				if (flags & FBF_EXPLICITANGLE)
				{
					pangle += spread_xy;
					slope += spread_z;
				}
				else
				{
					pangle += spread_xy * Random2[cabullet]() / 255.;
					slope += spread_z * Random2[cabullet]() / 255.;
				}

				int damage = damageperbullet;

				if (!(flags & FBF_NORANDOM))
					damage *= random[cabullet](1, 3);

				puff = LineAttack(pangle, range, slope, damage, 'Hitscan', pufftype, laflags, t);
			}
		}
		
		if(invoker.disableTracer || !puff) return;
		
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		let _puff = pmo.LineAttack(angle, 33, pitch+10, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 spawnPos = _puff.pos;
		
		let _tracer = pmo.SpawnPlayerMissile("BulletTracer");
		if(_tracer)
		{
			_tracer.A_SetAngle(_tracer.AngleTo(puff));
			_tracer.A_SetPitch(invoker.GetPitchTo(_tracer, puff));
			_tracer.Vel3DFromAngle(GetDefaultSpeed("BulletTracer"), _tracer.angle, _tracer.pitch);
		}
	}
	
	private void fireTracer()
	{
		let pmo = owner.player.mo;
		int hand = bOffhandWeapon;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		let _puff = pmo.LineAttack(angle, 33, pitch+10, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 spawnPos = _puff.pos;
		
		_puff = pmo.LineAttack(angle, PLAYERMISSILERANGE, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 destPos = _puff.pos;
		
		let _tracer = Actor.Spawn("BulletTracer", spawnPos);
		if(_tracer)
		{
			_tracer.A_SetAngle(_tracer.AngleTo(_puff));
			_tracer.A_SetPitch(GetPitchTo(_tracer, _puff));
			_tracer.Vel3DFromAngle(GetDefaultSpeed("BulletTracer"), _tracer.angle, _tracer.pitch);
		}
	}

	action void A_ReloadMag()
	{
		let pmo = invoker.owner;
		let magammo = pmo.CountInv("ChaingunMag");
		let totalammo = pmo.CountInv("Clip");
		int addammo;
		if(magammo == 0)
		{
			addammo = 60;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		else
		{
			addammo = 60 - magammo;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		magammo += addammo;
		totalammo = totalammo - addammo;
		totalammo = totalammo < 0 ? 0 : totalammo;
		pmo.SetInventory("ChaingunMag", magammo);
		if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Clip", totalammo);
		invoker.weaponUnloaded = false;
	}
	
	action void A_UnloadMag()
	{
		let pmo = invoker.owner;
		int maxammo = pmo.CountInv("backpack") ? 400 : 200;
		let magammo = pmo.CountInv("ChaingunMag");
		let totalammo = pmo.CountInv("Clip");
		if(totalammo + magammo > maxammo) return;
		pmo.SetInventory("ChaingunMag", 0);
		pmo.SetInventory("Clip", totalammo + magammo);
		invoker.weaponUnloaded = true;
	}
