
class RLPistol : DoomWeaponZ
{
 	Default
	{
	Weapon.SelectionOrder 1900;
	Weapon.AmmoUse1 0;
	Weapon.AmmoGive1 0;
	Weapon.AmmoUse2 1;
	Weapon.AmmoGive2 16;
	Weapon.AmmoType1 "Clip";
	Weapon.AmmoType2 "PistolMag";
	Obituary "$OB_MPPISTOL";
	+WEAPON.WIMPY_WEAPON
	Inventory.Pickupmessage "$PICKUP_PISTOL_DROPPED";
	Tag "$TAG_PISTOL";
	+WEAPON.NOALERT;
	+WEAPON.NOAUTOAIM;
	+WEAPON.NOAUTOFIRE;
	+WEAPON.AMMO_OPTIONAL;
	}
	
	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(owner.player.ReadyWeapon == self) checkOffhandGrab();
	}
	
	void checkOffhandGrab()
	{
		if(weaponModel < 2 || !(owner.player.offhandweapon is "RLOffhandFist") || owner.countinv("ImmersiveInteractToken"))
		{
			owner.player.mo.SetInventory("HandInteractToken", 0);
			return;
		}
		
		if(GetAge() % 10 != 0) return;
	
		let pmo = owner.player.mo;
		double handDist = level.Vec3Diff(owner.AttackPos, owner.OffhandPos).Length();
		if(handDist <= 4)
		{
			pmo.SetInventory("HandInteractToken", 1);
		}
		else
		{
			pmo.SetInventory("HandInteractToken", 0);
		}
	}
	
	action void A_CheckImmersiveReloadAction1()
	{
		// RSRL (RSQ Weapon Depot): measured hand reload, row 0; his touch test below runs outside VR or with rsrl_handreload 0
		if (RSRL_Reload.On(invoker)) { bool rsDone; Vector3 rsAt; [rsDone, rsAt] = RSRL_Reload.Gate(invoker, 0, !invoker.ImmersiveAmmoGrab || invoker.owner.CountInv("grabbedAmmo") > 0); invoker.showWeaponMarker(rsAt); if (rsDone) invoker.setReloadStep(2, 1); return; }
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 5, pitch-100, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 magReloadPos = _puff.pos;
		
		invoker.showWeaponMarker(magReloadPos);
		
		double handDistMag = level.Vec3Diff(magReloadPos, player.mo.OffhandPos).Length();
		if(handDistMag <= 3) invoker.setReloadStep(2, 1);
	}
	
	action void A_CheckImmersiveReloadAction2()
	{
		// RSRL (RSQ Weapon Depot): measured hand reload, row 1; his touch test below runs outside VR or with rsrl_handreload 0
		if (RSRL_Reload.On(invoker)) { bool rsDone; Vector3 rsAt; [rsDone, rsAt] = RSRL_Reload.Gate(invoker, 1, !invoker.ImmersiveAmmoGrab || invoker.owner.CountInv("grabbedAmmo") > 0); invoker.showWeaponMarker(rsAt); if (rsDone) invoker.setReloadStep(3); return; }
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 2.5, pitch+100, 0, "melee", "RLVRDUmmyPuff1", laflags);
		vector3 sliderPose = _puff.pos;
		
		invoker.showWeaponMarker(sliderPose);
		
		double handDistSlider = level.Vec3Diff(sliderPose, player.mo.OffhandPos).Length();
		if(handDistSlider <= 3) invoker.setReloadStep(3);
	}
	
	action void A_SpawnWeaponFlash()
	{
		let pmo = player.mo;
		int hand = invoker.bOffhandWeapon;
		Actor _puff;
		double angle = pmo.angle;
		double pitch = pmo.AimTarget() ? pmo.BulletSlope(null, ALF_PORTALRESTRICT) : pmo.pitch;
		int laflags = LAF_NOIMPACTDECAL | LAF_NOINTERACT | LAF_NORANDOMPUFFZ;
		if(hand) laflags |= LAF_ISOFFHAND;
		
		_puff = pmo.LineAttack(angle, 7, pitch, 0, "melee", "RLVRDUmmyPuff1", laflags);
		if(_puff) Actor.Spawn("RLVRPistolFlash", _puff.pos);
	}
	
	action void A_ReloadMag()
	{
		let pmo = invoker.owner;
		let magammo = pmo.CountInv("PistolMag");
		let totalammo = pmo.CountInv("Clip");
		int addammo;
		if(magammo == 0)
		{
			addammo = 15;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		else
		{
			addammo = 16 - magammo;
			addammo = addammo > totalammo ? totalammo : addammo;
		}
		magammo += addammo;
		totalammo = totalammo - addammo;
		totalammo = totalammo < 0 ? 0 : totalammo;
		pmo.SetInventory("PistolMag", magammo);
		if(!(player.cheats & CF_INFINITEAMMO)) pmo.SetInventory("Clip", totalammo);
		invoker.weaponUnloaded = false;
	}
	
	action void A_UnloadMag()
	{
		let pmo = invoker.owner;
		int maxammo = pmo.CountInv("backpack") ? 400 : 200;
		let magammo = pmo.CountInv("PistolMag");
		let totalammo = pmo.CountInv("Clip");
		if(totalammo + magammo > maxammo) return;
		pmo.SetInventory("PistolMag", 0);
		pmo.SetInventory("Clip", totalammo + magammo);
		invoker.weaponUnloaded = true;
	}
	
	States
	{
	Ready:
		PISG # 0 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_JumpIf(invoker.weaponUnloaded, "ReadyNoMag");
		#### # 0 A_JumpIf(invoker.owner.countinv("PistolMag") == 0, "ReadyEmpty");
		#### A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Loop;
	ReadyEmpty:
		PISG B 0 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyNoMag:
		PISG S 0 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	Deselect:
		PISG A 0 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 1 A_Lower (160);
		Loop;
	Select:
		PISG A 0 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 1 A_Raise (160);
		Loop;
	DryFire:
		#### # 0 A_JumpIf(invoker.autoReload, "Reload");
		PISG Z 0 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_JumpIf(invoker.weaponUnloaded, "DryFireNoMag");
		#### # 0 A_StartSound("weapon/dryfire", CHAN_WEAPON);
		#### ################## 1;
		Goto Ready;
	DryFireNoMag:
		#### Y 0 A_StartSound("weapon/dryfire", CHAN_WEAPON);
		#### ################## 1;
		Goto Ready;
	Fire:
		PISG # 0 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_JumpIfInventory("PistolMag", 1, 1);
		Goto DryFire;
		#### C 1 A_AlertMonsters;
	FireShot:
		#### # 0 A_StartSound("weapons/pistol/fire", CHAN_WEAPON);
		#### # 0 A_GunFlash("MuzzleFlash");
		#### # 0 A_SpawnWeaponFlash;
		#### # 0 A_JumpIf(invoker.disableTracer, 2);
		#### # 0 A_FireProjectile("Bullettracer", 0, 0, 0, 0, FPF_NOAUTOAIM, 0);
		#### E 1 A_FireBullets(1, 1, -1, 14, "EnhancedBulletPuff", FBF_NORANDOM);
		#### # 0 A_TakeInventory("PistolMag", 1);
		#### # 0 A_JumpIf(invoker.performanceMode, 3);
		#### # 0 A_FireProjectile("PistolCasingSpawner", 0, 0, 0, 0, FPF_NOAUTOAIM, 0);
		#### # 0 A_FireProjectile("WeaponBarrelSmoke",0,0,0,0,FPF_NOAUTOAIM,0);
		#### FG 1;
		#### # 0 A_JumpIf(invoker.owner.countinv("PistolMag") == 0, "EmptyFireDone");
		#### H 1;
		#### I 1 A_ReFire;
		Goto Ready;
	EmptyFireDone:
		#### # 0 A_JumpIf(invoker.autoReload, "Reload");
		#### # 1 A_StartSound("weapons/pistol/slideout", CHAN_WEAPON, CHANF_OVERLAP);
		Goto ReadyEmpty;
	Hold:
		#### D 18;
		#### # 0 A_JumpIf(invoker.owner.countinv("PistolMag") == 0, "EmptyFireDone");
		#### # 1 A_ReFire ("FireShot");
		Goto Ready;
	Reload:
		PISG ######## 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_JumpIfInventory("Clip", 1, 1);
		Goto ReloadNoAmmo;
		#### # 0 A_JumpIf(invoker.ImmersiveReload, "ImmersiveReload");
		#### # 0 A_JumpIf(invoker.weaponUnloaded, 17);
		PISG JJJJJ 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_StartSound("weapons/pistol/clipout", CHAN_WEAPON);
		#### # 0 { RLVRHapticHelper.BuzzHand(); } 						//haptic
		#### # 0 A_JumpIf(invoker.performanceMode, 2);
		#### K 1 A_FireProjectile("RLPistolMagSpawner", 0, 0, 0, -10, FPF_NOAUTOAIM, 0);
		PISG KKKKKKKLLLLLLLLLLLLMMMNN 1 {
			A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
			//A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_OffhandHaptic; 								//haptic
		#### N 1 A_StartSound("weapons/pistol/clipin", CHAN_WEAPON, CHANF_OVERLAP);
		#### OO 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		#### # 0 A_ReloadMag();
		#### P 1 A_StartSound("weapons/pistol/slideout", CHAN_WEAPON, CHANF_OVERLAP);
		PISG PPPPPPPPPQQ 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_OffhandHaptic; 								//haptic
		#### Q 1 A_StartSound("weapons/pistol/slidein", CHAN_WEAPON, CHANF_OVERLAP);
		PISG QQQQQQQQQ 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		Goto Ready;
	ReloadNoAmmo:
		#### ##### 1;
		Goto ReadyEmpty;
	Zoom:
	UnloadAmmo:
		PISG ######## 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_JumpIf(invoker.weaponUnloaded, "ReadyNoMag");
		PISG JJJJJ 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_StartSound("weapons/pistol/clipout", CHAN_WEAPON);
		#### # 0 { RLVRHapticHelper.BuzzHand(); } 						//haptic
		#### # 0 A_JumpIf(invoker.performanceMode, 2);
		#### K 1 A_FireProjectile("RLPistolMagSpawner", 0, 0, 0, -10, FPF_NOAUTOAIM, 0);
		#### # 0 A_UnloadMag;
		Goto ReadyNoMag;
	ImmersiveReload:
		#### # 0 A_JumpIf(invoker.weaponUnloaded, "WaitReload");
		PISG JJ 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_StartSound("weapons/pistol/clipout", CHAN_WEAPON);
		#### # 0 { RLVRHapticHelper.BuzzHand(); } 						//haptic
		#### # 0 A_JumpIf(invoker.performanceMode, 2);
		#### K 1 A_FireProjectile("RLPistolMagSpawner", 0, 0, 0, -10, FPF_NOAUTOAIM, 0);
	WaitReload:
		#### # 0 { invoker.setReloadStep(1); }
		PISG K 1
		{
			A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_CheckImmersiveReloadAction1();
		#### # 0 A_JumpIf(player.mo.countinv("ImmersiveReloadStep") == 2, "ImmersiveReloadMag");
		Loop;
	ImmersiveReloadMag:
		#### # 0 { invoker.hideWeaponMarker(); }
		PISG LLLMMMNN 1 {
			A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
			//A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_OffhandHaptic; 								//haptic
		#### N 1 A_StartSound("weapons/pistol/clipin", CHAN_WEAPON, CHANF_OVERLAP);
		#### OOOOOOOO 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		#### # 0 A_ReloadMag();
		#### # 0 { invoker.setReloadStep(0); }
	ImmersiveReloadWait2:
		PISG Y 1
		{
			A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		}
		#### # 0 A_CheckImmersiveReloadAction2();
		#### # 0 A_JumpIf(player.mo.countinv("ImmersiveReloadStep") == 3, "ImmersiveReloadDone");
		Loop;
	ImmersiveReloadDone:
		#### # 0 { invoker.hideWeaponMarker(); }
		#### # 0 A_StartSound("weapons/pistol/slideout", CHAN_WEAPON, CHANF_OVERLAP);
		PISG PPPPPQQ 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 A_OffhandHaptic; 								//haptic
		#### # 0 A_StartSound("weapons/pistol/slidein", CHAN_WEAPON, CHANF_OVERLAP);
		PISG QQQ 1 A_SetAltModel("PISP", "PS2H", "PS2P", "PSNH", "PSNP");
		#### # 0 { invoker.setReloadStep(0); }
		Goto Ready;
	MuzzleFlash:
		TNT1 A 0 A_OverlayFlags(OverlayID(), PSPF_RENDERSTYLE | PSPF_FORCEALPHA, true);
		TNT1 A 0 A_OverlayRenderstyle(OverlayID(), STYLE_Add);
		TNT1 A 0 A_OverlayAlpha(OverlayID(), invoker.flashAlpha);
		TNT1 A 0 A_JumpIf(invoker.showFLashSprite == false, 2);
		TNT1 A 0 A_Jump(256, "MuzzleFlash1", "MuzzleFlash2", "MuzzleFlash3", "MuzzleFlash4", "MuzzleFlash5");
		TNT1 A 0;
		stop;
	MuzzleFlash1:
		PISF A 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash2:
		PISF B 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash3:
		PISF C 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash4:
		PISF D 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
	MuzzleFlash5:
		PISF E 2 Bright A_Light1;
		TNT1 A 5 Bright A_Light1;
		Goto LightDone;
 	Spawn:
		PIST A -1;
		Stop;
	PlaceHolder:
		PISP A 1;
		PS2H A 1;
		PS2P A 1;
		PSNH A 1;
		PSNP A 1;
		Stop;
	}
