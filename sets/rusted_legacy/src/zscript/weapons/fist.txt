
class RLFist : DoomWeaponZ
{
	Default
	{
	Weapon.SelectionOrder 3701;
	Weapon.KickBack 100;
	Obituary "$OB_MPFIST";
	Tag "$TAG_FIST";
	+WEAPON.WIMPY_WEAPON
	+WEAPON.MELEEWEAPON
	+WEAPON.NOHANDSWITCH
	+WEAPON.NOALERT
	}
	
	States
	{
	Ready:
		TNT1 A 0;
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Loop;
	Deselect:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### A 1 A_Lower (160);
		Loop;
	Select:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### A 1 A_Raise (160);
		Loop;
	Fire:
	Hold:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### BC 1;
		#### D 1 A_Punch;
		#### EF 1;
		#### F 5;
		#### EDCB 1;
		#### B 8 A_ReFire;
		Goto Ready;
	Reload:
		#### # 10;
		Goto Ready;
	PlaceHolder:
		FSTP A 1;
		Stop;
	}
}

class RLOffhandFist : RLFist
{
	bool shieldMode;
	bool shieldIdle;
	
	Default
	{
	Weapon.SelectionOrder 3702;
	+WEAPON.OFFHANDWEAPON
	}
	
	override void Tick()
	{
		if(!owner || !owner.player || owner.health < 1) return;
		
		Super.Tick();
		if(!owner.player.OffhandWeapon)
		{
			owner.player.PendingWeapon = self;
		}
		
		checkForProtectorShield();
	}
	
	action void A_CheckAmmoGrabbed()
	{
		if(player.mo.countinv("ImmersiveReloadStep") == 1)
		{
			if(invoker.ImmersiveAmmoGrab && !player.mo.countinv("grabbedAmmo")) return;
		
			let currentWeap = player.readyWeapon;
			if(currentWeap is "RLPistol") 		SetWeaponState("ReadyPistolAmmo");
			if(currentWeap is "RLShotgun") 		SetWeaponState("ReadyShottyAmmo");
			if(currentWeap is "HellShotgun") 	SetWeaponState("ReadyHellShottyAmmo");
			if(currentWeap is "RLSuperShotgun") 	SetWeaponState("ReadySSGAmmo");
			if(currentWeap is "RLChaingun") 	SetWeaponState("ReadyChaingunAmmo");
			if(currentWeap is "RLRocketLauncher") 	SetWeaponState("ReadyRocketAmmo");
			if(currentWeap is "RLPlasmaRifle") 	SetWeaponState("ReadyPlasmaAmmo");
			if(currentWeap is "RLBFG9000") 		SetWeaponState("ReadyBFGAmmo");
			if(currentWeap is "RLHeatwave") 	SetWeaponState("ReadyHeatAmmo");
			if(currentWeap is "RLIncinerator") 	SetWeaponState("ReadyFlameAmmo");
		}
	}
	
	action void A_CheckShieldMode()
	{
		if(invoker.shieldReturned) invoker.shieldMode = invoker.shieldIdle = false;
	}
	
	States
	{
	Ready:
		TNT1 A 0;
		#### # 0 A_CheckAmmoGrabbed();
		#### A 0 A_JumpIf(invoker.owner.countinv("ImmersiveReloadStep") == 6, 3);
		#### A 0 A_JumpIfInventory("HandInteractToken", 1, "NoHand");
		#### A 0 A_JumpIf(invoker.owner.countinv("ImmersiveReloadStep") > 1, "NoHand");
		#### A 0 A_JumpIf(invoker.shieldMode, "ReadyShield");
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### A 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Loop;
	ReadyPistolAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### Q 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyShottyAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### R 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyHellShottyAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### P 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadySSGAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### S 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyChaingunAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### T 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyRocketAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### U 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyPlasmaAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### V 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyBFGAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### W 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyHeatAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### X 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	ReadyFlameAmmo:
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### Y 1 A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
		Goto Ready;
	Fire:
		#### A 0 A_JumpIfInventory("HandInteractToken", 1, "NoHand");
		#### A 0 A_JumpIfInventory("ImmersiveInteractToken", 1, "NoHand");
		#### A 0 A_JumpIf(invoker.shieldMode, "ShieldSawActivate");
		FIST # 0 A_SetAltModel("FSTP", "----", "----", "----", "FSTP");
		#### BC 1;
		#### D 1 A_Punch;
		#### EF 1;
		#### F 5;
		#### EDCB 1;
		#### B 8 A_ReFire;
		Goto Ready;
	AltFire:
		#### # 10
		{
			if(invoker.shieldSaw)
			{
				invoker.shieldMode = !invoker.shieldMode;
				invoker.shieldIdle = invoker.shieldMode;
			}
		}
		Goto Ready;
	Hold:
		#### A 0 A_JumpIf(invoker.shieldMode, "ShieldSawAttack");
		Goto Fire;
	Reload:
		#### # 1;
		Goto Ready;
	NoHand:
		TNT1 A 1
		{
			A_WeaponReady(WRF_ALLOWRELOAD | WRF_ALLOWZOOM);
			A_CheckShieldMode();
		}
		Goto Ready;
	}
}
