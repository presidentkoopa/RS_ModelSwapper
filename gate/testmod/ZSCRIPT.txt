// THE STAND-IN FOR "THEIR CODE". A pistol written the way a mod writes one, owing nothing to
// ModelSwapper: its own Reload state, its own ammo names (TMMag in the gun, Clip in reserve),
// and its own rule for what a reload does. ModelSwapper must find its reload, hand it to the
// player's hands, and give it back so THIS code decides the ammo. Test only; never shipped.
class TMMag : Ammo
{
	Default { Inventory.MaxAmount 12; Ammo.BackpackMaxAmount 12; +INVENTORY.IGNORESKILL; }
}

class TMPistol : Weapon replaces Pistol
{
	Default
	{
		Weapon.SlotNumber 2;
		Weapon.AmmoType1 "TMMag";
		Weapon.AmmoUse1 1;
		Weapon.AmmoGive1 12;
		Weapon.AmmoType2 "Clip";
		Weapon.AmmoGive2 50;
		Weapon.SelectionOrder 1900;
		+WEAPON.AMMO_OPTIONAL
		Tag "Test Pistol";
	}
	States
	{
	Ready:
		PISG A 1 A_WeaponReady(WRF_ALLOWRELOAD);
		Loop;
	Deselect:
		PISG A 1 A_Lower;
		Loop;
	Select:
		PISG A 1 A_Raise;
		Loop;
	Fire:
		PISG A 0 A_JumpIfNoAmmo("Reload");
		PISG B 6 A_FirePistol;
		PISG C 4;
		PISG B 5 A_ReFire;
		Goto Ready;
	Reload:
		PISG A 0 A_JumpIf(CountInv("TMMag") >= 12 || CountInv("Clip") <= 0, "Ready");
		PISG B 8 A_Log("TM reload: the mod's own reload runs");
		PISG C 8;
		PISG A 0
		{
			int need = 12 - CountInv("TMMag");
			int have = min(need, CountInv("Clip"));
			A_TakeInventory("Clip", have);
			A_GiveInventory("TMMag", have);
			A_Log(String.Format("TM reload: mag %d, reserve %d", CountInv("TMMag"), CountInv("Clip")));
		}
		Goto Ready;
	Spawn:
		PIST A -1;
		Stop;
	}
}

// A pump shotgun written the way mods write them: a tube (TMShells) and a reserve (Shell),
// and a reload that is a loop putting ONE shell in per pass, checking for room each time.
// ModelSwapper must pace this loop with the player's hand, shell by shell, and leave every
// decision -- room, cost, sound -- to this code.
class TMShells : Ammo
{
	Default { Inventory.MaxAmount 8; Ammo.BackpackMaxAmount 8; +INVENTORY.IGNORESKILL; }
}

class TMShotgun : Weapon replaces Shotgun
{
	Default
	{
		Weapon.SlotNumber 3;
		Weapon.AmmoType1 "TMShells";
		Weapon.AmmoUse1 1;
		Weapon.AmmoGive1 8;
		Weapon.AmmoType2 "Shell";
		Weapon.AmmoGive2 20;
		Weapon.SelectionOrder 1300;
		+WEAPON.AMMO_OPTIONAL
		Tag "Test Shotgun";
	}
	States
	{
	Ready:
		SHTG A 1 A_WeaponReady(WRF_ALLOWRELOAD);
		Loop;
	Deselect:
		SHTG A 1 A_Lower;
		Loop;
	Select:
		SHTG A 1 A_Raise;
		Loop;
	Fire:
		SHTG A 0 A_JumpIfNoAmmo("Reload");
		SHTG A 3 A_FireShotgun;
		SHTG BC 5;
		SHTG D 4;
		SHTG CB 5;
		SHTG A 3;
		SHTG A 7 A_ReFire;
		Goto Ready;
	Reload:
		SHTG A 0 A_JumpIf(CountInv("TMShells") >= 8 || CountInv("Shell") <= 0, "Ready");
		SHTG A 0 A_Log("TM shotgun: the mod's own reload starts");
	LoadOne:
		SHTG B 6;
		SHTG C 6
		{
			A_TakeInventory("Shell", 1);
			A_GiveInventory("TMShells", 1);
			A_Log(String.Format("TM shotgun: a shell in, tube %d, reserve %d", CountInv("TMShells"), CountInv("Shell")));
		}
		SHTG A 0 A_JumpIf(CountInv("TMShells") >= 8 || CountInv("Shell") <= 0, "LoadDone");
		Loop;
	LoadDone:
		SHTG A 0 A_Log(String.Format("TM shotgun: reload done, tube %d", CountInv("TMShells")));
		SHTG BA 5;
		Goto Ready;
	Spawn:
		SHOT A -1;
		Stop;
	}
}

// A break-open double barrel written the way mods write them: two barrels (TMBarrels) and a
// reserve (Shell), one reload that fills both. ModelSwapper must hold it through the hands'
// open-load-close and then hand it to THIS reload.
class TMBarrels : Ammo
{
	Default { Inventory.MaxAmount 2; Ammo.BackpackMaxAmount 2; +INVENTORY.IGNORESKILL; }
}

class TMSSG : Weapon replaces SuperShotgun
{
	Default
	{
		Weapon.SlotNumber 3;
		Weapon.AmmoType1 "TMBarrels";
		Weapon.AmmoUse1 2;
		Weapon.AmmoGive1 2;
		Weapon.AmmoType2 "Shell";
		Weapon.AmmoGive2 20;
		Weapon.SelectionOrder 400;
		+WEAPON.AMMO_OPTIONAL
		Tag "Test Super Shotgun";
	}
	States
	{
	Ready:
		SHT2 A 1 A_WeaponReady(WRF_ALLOWRELOAD);
		Loop;
	Deselect:
		SHT2 A 1 A_Lower;
		Loop;
	Select:
		SHT2 A 1 A_Raise;
		Loop;
	Fire:
		SHT2 A 0 A_JumpIfNoAmmo("Reload");
		SHT2 A 3;
		SHT2 A 7 A_FireShotgun2;
		SHT2 B 7;
		Goto Ready;
	Reload:
		SHT2 A 0 A_JumpIf(CountInv("TMBarrels") >= 2 || CountInv("Shell") <= 0, "Ready");
		SHT2 C 7 A_Log("TM ssg: the mod's own reload runs");
		SHT2 D 7;
		SHT2 E 7
		{
			int have = min(2 - CountInv("TMBarrels"), CountInv("Shell"));
			A_TakeInventory("Shell", have);
			A_GiveInventory("TMBarrels", have);
			A_Log(String.Format("TM ssg: barrels %d, reserve %d", CountInv("TMBarrels"), CountInv("Shell")));
		}
		SHT2 F 5;
		Goto Ready;
	Spawn:
		SGN2 A -1;
		Stop;
	}
}
