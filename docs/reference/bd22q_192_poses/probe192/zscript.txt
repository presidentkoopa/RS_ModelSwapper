version "4.10"
// TEST ONLY (gallery_192): logs the held weapon, its state and the two-hand flag once a second.
class G192_Probe : EventHandler
{
	override void WorldTick()
	{
		if (level.maptime % 35 != 0) return;
		let p = players[consoleplayer];
		if (!p.mo) return;
		let w = p.ReadyWeapon;
		let ps = p.FindPSprite(PSP_WEAPON);
		String st = (p.WeaponState & WF_WEAPONREADY) ? "READY" : "busy";
		let s1 = p.FindPSprite(2101);
		let s0 = p.FindPSprite(2100);
		Console.Printf("G192 %d ready %s pending %s state %s two %d sup %s gun %s refire %d", level.maptime,
			w ? String.Format("%s", w.GetClassName()) : "none",
			p.PendingWeapon && p.PendingWeapon != WP_NOCHANGE ? String.Format("%s", p.PendingWeapon.GetClassName()) : "-",
			st, (p.WeaponState & WF_TWOHANDSTABILIZED) ? 1 : 0,
			s1 && s1.CurState ? String.Format("%d", s1.Sprite) : "none",
			s0 && s0.CurState ? String.Format("%d", s0.Sprite) : "none", p.refire);
	}
}
